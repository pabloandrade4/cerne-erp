  /* ================= ADS — Product Ads real do Mercado Livre =================
     Ativado em 25/08/2026, corrigido em 25/08/2026 (ver docs/04-alteracoes.md).
     Duas fontes bem separadas (ver lib/ads.js): investimento/vendas
     atribuídas/receita atribuída vêm SEMPRE da API de Advertising do
     Mercado Livre (nunca calculados aqui); faturamento real/margem "antes
     do Ads" vêm da MESMA fonte única de Pedidos/DRE/Financeiro
     (lib/relatorioVendas.js). Nunca inventa valor: quando a API de Ads não
     está disponível pra uma conta, os campos aparecem como "Pendente de
     sincronização" — nunca um número estimado. Empresa e período vêm do
     filtro do header (window.CerneFiltro); loja é um filtro só desta tela.

     IMPORTANTE: a API de Advertising do Mercado Livre não identifica quais
     PEDIDOS pertencem à publicidade — só devolve totais agregados por
     anúncio/dia. Por isso a tabela por anúncio é dividida em DUAS visões
     separadas, nunca misturadas numa só: "Performance atribuída Mercado
     Ads" (só o que o Mercado Livre atribui) e "Resultado real do SKU após

  (function(){
    const state = { loading:false, error:null, data:null, contaId:'', sort:'faturamento', sincronizando:false, salvandoMargemMinima:false, campanhasExpandidas:{} };

    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+message+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 3200);
    }
    function fmtDateTime(iso){ if(!iso) return null; try{ return new Date(iso).toLocaleString('pt-BR'); }catch(e){ return null; } }
    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function money(v){ return v===null||v===undefined ? null : 'R$ ' + Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2}); }
    function pendenteSync(){ return '<span style="color:var(--text-faint);font-style:italic;" title="Este dado depende da API de Publicidade do Mercado Livre, que ainda não retornou (ou a conta não tem acesso a Product Ads).">Pendente de sincronização</span>'; }
    function pendenteCusto(){ return '<span style="color:var(--text-faint);font-style:italic;" title="Falta cadastrar o custo de algum SKU deste anúncio em Produtos.">pendente</span>'; }
    function moneyOrPend(v){ return v===null||v===undefined ? pendenteSync() : money(v); }
    function roasFmt(v){ return v===null||v===undefined ? pendenteSync() : Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2})+'x'; }
    function pctFmt(v){ return v===null||v===undefined ? pendenteSync() : Number(v).toLocaleString('pt-BR',{minimumFractionDigits:1, maximumFractionDigits:1})+'%'; }
    function intFmt(v){ return v===null||v===undefined ? pendenteSync() : Number(v).toLocaleString('pt-BR'); }
    function fmtDiaCurto(diaISO){ const [, m, d] = diaISO.split('-'); return d+'/'+m; }

    async function apiAds(empresaId, periodo, contaId, desde, ate){
      const params = new URLSearchParams({ empresaId, periodo });
      if(contaId) params.set('contaId', contaId);
      if(periodo === 'personalizado'){ if(desde) params.set('desde', desde); if(ate) params.set('ate', ate); }
      const res = await fetch('/api/ads?'+params.toString());
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok) throw new Error((body && body.error) || 'Não foi possível carregar os dados de Ads.');
      return body;
    }

    // ---- IA de Ads e Performance — config da margem mínima (mesmo padrão de
    // GET/PUT /api/promocoes/config) — o valor atual vem sempre junto da
    // resposta de /api/ads (campo margemMinimaPct), então só precisamos do
    // PUT aqui pra salvar; depois de salvar recarregamos /api/ads pra
    // reclassificar tudo (anúncio e campanha) com o novo mínimo. ----
    async function apiSalvarMargemMinimaAds(empresaId, margemMinimaPct){
      const res = await fetch('/api/ads/config-ia', { method:'PUT', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ empresaId, margemMinimaPct }) });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok) throw new Error((body && (body.error||(body.errors&&JSON.stringify(body.errors)))) || 'Falha ao salvar a margem mínima.');
      return body;
    }

    async function salvarMargemMinima(valorStr){
      const filtro = window.CerneFiltro.state;
      const valor = Number(valorStr);
      if(!Number.isFinite(valor) || valor < 0 || valor > 100){
        toast('Informe um percentual entre 0 e 100.', 'error'); return;
      }
      state.salvandoMargemMinima = true; render();
      try{
        await apiSalvarMargemMinimaAds(filtro.empresaId, valor);
        toast('Margem mínima da IA de Ads atualizada.');
        await loadAds();
      }catch(err){
        toast(err.message || 'Falha ao salvar a margem mínima.', 'error');
      }finally{
        state.salvandoMargemMinima = false; render();
      }
    }

    async function apiSincronizarAds(){
      const res = await fetch('/api/ads/sincronizar', { method:'POST' });
      const body = await res.json().catch(()=>null);
      if(!res.ok) throw new Error((body && body.error) || 'Falha ao sincronizar Ads.');
      return body;
    }

    async function sincronizarAgora(){
      state.sincronizando = true; render();
      try{
        await apiSincronizarAds();
        toast('Sincronização de Ads concluída.');
        await loadAds();
      }catch(err){
        toast(err.message || 'Falha ao sincronizar Ads.', 'error');
      }finally{
        state.sincronizando = false; render();
      }
    }

    async function loadAds(){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId){ state.data = null; state.loading = false; render(); return; }
      state.loading = true; state.error = null; render();
      try{ state.data = await apiAds(filtro.empresaId, filtro.periodo, state.contaId, filtro.desdePersonalizado, filtro.atePersonalizado); }
      catch(err){ state.error = err.message; state.data = null; }
      state.loading = false; render();
    }

    function load(){ loadAds(); }

    window.CerneFiltro.onChange(()=>{
      const page = document.getElementById('page-ads');
      if(page && page.classList.contains('active')) loadAds();
      else render();
    });

    }

    // ---- Gráfico diário: Investimento Ads x Receita atribuída (SVG simples, mesmo desenho de Visão Geral) ----
    function chartAdsHTML(diario){
      if(!diario || !diario.length){
        return '<div class="chart-canvas h-lg">' + chartGridLines(4)
          + chartEmptyState('lineChart', 'Sem série diária de Ads disponível para este período (a API de Publicidade pode não ter retornado dado, ou nenhuma conta tem acesso a Product Ads).')
        + '</div>';
      }
      const n = diario.length;
      const W = Math.max(n*10, 10), H = 100, topPad = 8, botPad = 10;
      const usable = H - topPad - botPad;
      const vals = diario.flatMap(d => [d.investimento||0, d.receitaAtribuida||0]);
      const maxV = Math.max(0, ...vals);
      const minV = Math.min(0, ...vals);
      const range = (maxV - minV) || 1;
      const yFor = v => topPad + usable * (1 - (v - minV) / range);
      const baseY = yFor(0);

      const barW = 6;
      let bars = '';
      let points = [];
      diario.forEach((d, i) => {
        const cx = i*10 + 5;
        const invest = d.investimento || 0;
        const y1 = yFor(invest);
        const barY = Math.min(y1, baseY), barH = Math.max(Math.abs(y1-baseY), 0.6);
        const dataLabel = fmtDiaCurto(d.data)+' — Investimento: '+(d.investimento===null?'pendente':money(d.investimento))+' · Receita atribuída: '+(d.receitaAtribuida===null?'pendente':money(d.receitaAtribuida));
        bars += '<rect x="'+(cx-barW/2)+'" y="'+barY+'" width="'+barW+'" height="'+barH+'" rx="1" fill="var(--copper)" fill-opacity="0.85"><title>'+escapeHtml(dataLabel)+'</title></rect>';
        points.push(cx+','+yFor(d.receitaAtribuida||0));
      });
      const line = '<polyline points="'+points.join(' ')+'" fill="none" stroke="var(--teal)" stroke-width="2" vector-effect="non-scaling-stroke" stroke-linejoin="round" stroke-linecap="round"/>';
      const dots = diario.map((d,i) => {
        const cx = i*10+5, cy = yFor(d.receitaAtribuida||0);
        const dataLabel = fmtDiaCurto(d.data)+' — Receita atribuída: '+(d.receitaAtribuida===null?'pendente':money(d.receitaAtribuida));
        return '<circle cx="'+cx+'" cy="'+cy+'" r="1.6" fill="var(--teal)" vector-effect="non-scaling-stroke"><title>'+escapeHtml(dataLabel)+'</title></circle>';
      }).join('');
      const baseline = '<line x1="0" y1="'+baseY+'" x2="'+W+'" y2="'+baseY+'" stroke="var(--border-strong)" stroke-width="1" vector-effect="non-scaling-stroke"/>';

      const labelStep = Math.max(1, Math.ceil(n/9));
      const labels = diario.map((d,i) => '<span>'+((i%labelStep===0 || i===n-1) ? escapeHtml(fmtDiaCurto(d.data)) : '')+'</span>').join('');

      return '<div class="chart-canvas" style="position:relative;height:200px;padding:0;">'
        + '<svg class="simple-chart-svg" viewBox="0 0 '+W+' '+H+'" preserveAspectRatio="none" style="height:100%;">'
          + baseline + bars + line + dots
        + '</svg>'
      + '</div>'
      + '<div class="simple-chart-xlabels" style="grid-template-columns:repeat('+n+',1fr);">' + labels + '</div>';
    }

    // ---- Ordenação da tabela por anúncio (aplicada às duas visões, pra manter a mesma ordem nas duas) ----
    function valOr(l, campo, seAusente){ const v = l[campo]; return v===null||v===undefined ? seAusente : v; }
    function comparadorDe(sort){
      if(sort==='lucro') return (a,b)=> valOr(b,'margemDepoisDoAds',-Infinity) - valOr(a,'margemDepoisDoAds',-Infinity);
      if(sort==='prejuizo') return (a,b)=> valOr(a,'margemDepoisDoAds',Infinity) - valOr(b,'margemDepoisDoAds',Infinity);
      if(sort==='gastoAds') return (a,b)=> valOr(b,'investimento',-Infinity) - valOr(a,'investimento',-Infinity);
      if(sort==='roas') return (a,b)=> valOr(b,'roas',-Infinity) - valOr(a,'roas',-Infinity);
      if(sort==='acos') return (a,b)=> valOr(b,'acos',-Infinity) - valOr(a,'acos',-Infinity); // maior ACOS = pior
      return (a,b)=> (b.faturamentoReal||0) - (a.faturamentoReal||0); // 'faturamento' (padrão)
    }

    // ---- IA de Ads e Performance — Fase A (14/09/2026): classificação em 5
    // níveis (mesmo espírito de badgeClassificacaoHTML da IA de Promoções),
    // substituindo o antigo status binário Lucrativo/Prejuízo. O backend
    // (lib/ia/adsMotor.js) já manda classificacaoCodigo/Label/Emoji prontos
    // pra cada anúncio e pra cada campanha — o front nunca decide sozinho. ----
    const CLASSIFICACAO_COR_ADS = {
      pausar:'var(--danger)', ajustar:'var(--warning)', manter:'var(--success)',
      escalar:'var(--copper)', dados_insuficientes:'var(--text-faint)',
    };

        ? 'Falta custo do produto ou métrica de Ads ainda não sincronizada.'
        : '';
      return '<span style="display:inline-flex;align-items:center;gap:5px;font-weight:700;font-size:12.5px;color:'+cor+';white-space:nowrap;" title="'+escapeHtml(titulo)+'">'
        + (l.classificacaoEmoji||'')+' '+escapeHtml(l.classificacaoLabel||'—')
      + '</span>';
    }

          + anuncioCellHTML(l)
          + '<td>'+escapeHtml(l.loja||'—')+'</td>'
          + '<td class="mono">'+(l.semVendaReal ? '<span style="color:var(--text-faint);">sem venda no período</span>' : money(l.faturamentoReal))+'</td>'
          + '<td class="mono">'+moneyOrPend(l.custoAds)+'</td>'
          + '<td class="mono">'+pctFmt(l.tacos)+'</td>'
          + '<td class="mono">'+(l.semVendaReal ? '<span style="color:var(--text-faint);">—</span>' : (l.margemAntesDoAds===null ? pendenteCusto() : money(l.margemAntesDoAds)))+'</td>'
          + '<td class="mono" style="'+(l.margemDepoisDoAds!==null && l.margemDepoisDoAds!==undefined?(l.margemDepoisDoAds>=0?'color:var(--success);':'color:var(--danger);'):'')+'">'+(l.margemDepoisDoAds===null||l.margemDepoisDoAds===undefined ? (l.semMetricasAds ? pendenteSync() : pendenteCusto()) : money(l.margemDepoisDoAds))+'</td>'
          + '<td class="mono">'+pctFmt(l.margemDepoisDoAdsPct)+'</td>'
          + '<td>'+classificacaoBadgeHTML(l)+'</td>'
        + '</tr>').join('')
        + '</tbody></table></div></div>';
    }

    function render(){
      const root = document.getElementById('adsRoot');
      if(!root) return;
      const filtro = window.CerneFiltro.state;

      if(!filtro.loadingEmpresas && !filtro.empresas.length){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
          + '<h2>Nenhuma empresa ativa</h2>'
          + '<p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> para ver Ads aqui.</p></div>';
        return;
      }

      let html = '<div class="field-hint" style="margin-bottom:12px;max-width:900px;">Investimento, cliques, impressões, CPC, vendas/receita atribuída, ROAS e ACOS vêm direto da API de Publicidade (Product Ads) do Mercado Livre — nunca um valor estimado pelo ERP. A API do Mercado Livre não informa quais pedidos pertencem à publicidade, só totais agregados — por isso as tabelas abaixo mostram duas visões SEPARADAS, nunca misturadas: <strong>Performance atribuída Mercado Ads</strong> (só o que a API atribui) e <strong>Resultado real do SKU após Ads</strong> (todas as vendas reais daquele SKU/anúncio, que podem incluir venda orgânica, menos o investimento em Ads).</div>';

      if(state.data){
        html += '<div class="toolbar">'
          + '<div style="display:flex;gap:10px;flex-wrap:wrap;align-items:center;">'
            + '<div class="empresa-picker" style="margin-bottom:0;"><select id="adsLojaSelect">'
              + '<option value="">Todas as lojas</option>'
              + state.data.lojas.map(l=>'<option value="'+l.id+'" '+(String(l.id)===String(state.contaId)?'selected':'')+'>'+escapeHtml(l.nickname||('Conta '+l.id))+'</option>').join('')
            + '</select></div>'
            + '<div class="empresa-picker" style="margin-bottom:0;"><select id="adsSortSelect">'
              + OPCOES_ORDENACAO.map(o=>'<option value="'+o.valor+'" '+(o.valor===state.sort?'selected':'')+'>Ordenar por: '+escapeHtml(o.label)+'</option>').join('')
            + '</select></div>'
            + '<button class="primary-btn enabled" id="btnSincronizarAds" type="button" '+(state.sincronizando?'disabled':'')+'>'+svg('refreshCw')+'<span>'+(state.sincronizando?'Sincronizando…':'Sincronizar agora')+'</span></button>'
            + '<div style="display:flex;align-items:center;gap:6px;" title="Abaixo deste percentual de margem após Ads, a IA classifica o anúncio/campanha como AJUSTAR ou PAUSAR.">'
              + '<label for="adsMargemMinimaInput" class="field-hint" style="margin:0;white-space:nowrap;">Margem mínima da IA (%)</label>'
              + '<input type="number" id="adsMargemMinimaInput" min="0" max="100" step="0.1" value="'+escapeHtml(String(state.data.margemMinimaPct!==undefined && state.data.margemMinimaPct!==null ? state.data.margemMinimaPct : 10))+'" style="width:70px;">'
              + '<button class="ghost-btn" id="adsSalvarMargemMinima" type="button" style="cursor:pointer;opacity:1;" '+(state.salvandoMargemMinima?'disabled':'')+'>'+svg('check')+'<span>'+(state.salvandoMargemMinima?'Salvando…':'Salvar')+'</span></button>'
            + '</div>'
          + '</div>'
          + (state.data.sincronizacaoAutomatica ? '<span class="panel-note" style="font-size:12px;">Sincronização automática em segundo plano'+(state.data.sincronizacaoAutomatica.ultimaExecucaoEm ? (' — última execução: '+escapeHtml(fmtDateTime(state.data.sincronizacaoAutomatica.ultimaExecucaoEm))) : ' ainda não rodou um ciclo completo')+'.</span>' : '')
        + '</div>';

        if(state.data.situacaoPorConta && state.data.situacaoPorConta.some(s=>!s.disponivel)){
          html += '<div class="table-card" style="padding:12px 16px;margin-bottom:14px;border-left:3px solid var(--warning);">'
            + state.data.situacaoPorConta.filter(s=>!s.disponivel).map(s=>
              '<div style="font-size:12.5px;color:var(--text-muted);display:flex;gap:8px;align-items:flex-start;margin:4px 0;">'
                + '<span style="color:var(--warning);flex-shrink:0;">'+svg('alertCircle')+'</span>'
                + '<span><strong>'+escapeHtml(s.loja||('Conta '+s.contaId))+'</strong>: '+escapeHtml(s.mensagem||'Dados de Ads indisponíveis para esta conta.')+'</span>'
              + '</div>'
            ).join('')
          + '</div>';
        }

        if(state.data.cards) html += cardsHTML(state.data.cards);
        html += chartAdsCardHTML(state.data);
      }

      if(state.loading || filtro.loadingEmpresas){
        html += '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando Ads…</span></div></div>';
      } else if(state.error){
        html += '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
          + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p></div>';
      } else if(state.data && state.data.semConta){
        html += '<div class="empty-panel"><div class="empty-icon">'+svg('store')+'</div>'
          + '<h2>Nenhuma conta do Mercado Livre conectada</h2>'
          + '<p class="lead">Conecte uma conta do Mercado Livre em <strong>Marketplaces</strong> para ver Ads aqui.</p></div>';
      } else if(state.data && !state.data.linhas.length){
        html += '<div class="empty-panel"><div class="empty-icon">'+svg('megaphone')+'</div>'
          + '<h2>Nenhum anúncio com venda ou dado de Ads no período</h2>'
          + '<p class="lead">Não há vendas nem métricas de publicidade para "'+escapeHtml((state.data.periodo&&state.data.periodo.label)||'')+'". Escolha outro período ou verifique a sincronização de pedidos.</p></div>';
      } else if(state.data){
        const linhasOrdenadas = state.data.linhas.slice().sort(comparadorDe(state.sort));
        html += '<div class="panel-title" style="margin:18px 0 8px;"><span class="panel-title-ic">'+svg('barChart')+'</span>Resumo por Campanha (IA de Ads e Performance)</div>'
          + '<div class="field-hint" style="margin-bottom:8px;">Soma dos números reais de cada anúncio (nunca recalculados) agrupados por campanha, pra identificar rapidamente campanhas boas e ruins. Clique em qualquer campanha pra ver os anúncios dela por baixo, igual no Mercado Livre. 🔴 PAUSAR = resultado negativo após Ads · 🟠 AJUSTAR = positivo mas abaixo da margem mínima configurada acima · 🟡 MANTER = margem saudável · 🟢 ESCALAR = margem folgada, dá pra investir mais · ⚠️ DADOS INSUFICIENTES = falta custo do produto ou métrica de Ads. Anúncios sem campanha (venda orgânica) não entram aqui.</div>'
          + tabelaCampanhasHTML(state.data.campanhas, linhasOrdenadas)
          + '<div class="panel-title" style="margin:22px 0 8px;"><span class="panel-title-ic">'+svg('megaphone')+'</span>Performance atribuída Mercado Ads</div>'
          + '<div class="field-hint" style="margin-bottom:8px;">Investimento, vendas e receita que o Mercado Livre atribui diretamente à publicidade — dado real da API, nunca estimado.</div>'
          + tabelaAtribuidaHTML(linhasOrdenadas)
          + '<div class="panel-title" style="margin:22px 0 8px;"><span class="panel-title-ic">'+svg('wallet')+'</span>Resultado real do SKU após Ads</div>'
          + '<div class="field-hint" style="margin-bottom:8px;">Margem de contribuição real de <strong>todas</strong> as vendas daquele SKU/anúncio no período (venda − taxas/comissões − frete do vendedor − imposto − custo do produto), menos o investimento em Ads. <strong>Pode incluir vendas orgânicas</strong> (não geradas pela publicidade) — isso NÃO é "lucro gerado pelo Ads", é o resultado financeiro real do SKU já descontando o investimento em Ads.</div>'
          + tabelaResultadoRealHTML(linhasOrdenadas)
          + '<div class="field-hint" style="margin-top:6px;">"Pendente de sincronização" = a API de Publicidade do Mercado Livre ainda não retornou esse dado (ou a conta não tem Product Ads habilitado) — nunca um valor estimado. "pendente" = falta cadastrar o custo de algum SKU deste anúncio em <strong>Produtos</strong>.</div>';
      }

      root.innerHTML = html;

      const selLoja = document.getElementById('adsLojaSelect');
      if(selLoja) selLoja.addEventListener('change', ()=>{ state.contaId = selLoja.value; loadAds(); });
      const selSort = document.getElementById('adsSortSelect');
      if(selSort) selSort.addEventListener('change', ()=>{ state.sort = selSort.value; render(); });
      const btnSync = document.getElementById('btnSincronizarAds');
      if(btnSync) btnSync.addEventListener('click', sincronizarAgora);
      const btnMargemMinima = document.getElementById('adsSalvarMargemMinima');
      const inputMargemMinima = document.getElementById('adsMargemMinimaInput');
      if(btnMargemMinima && inputMargemMinima) btnMargemMinima.addEventListener('click', ()=> salvarMargemMinima(inputMargemMinima.value));
      root.querySelectorAll('.campanhaExpandToggle').forEach(tr=>{
        tr.addEventListener('click', ()=>{
          const chave = tr.dataset.chave;
          state.campanhasExpandidas[chave] = !state.campanhasExpandidas[chave];
          render();
        });
      });
    }

    window.Ads = { load };
  })();
