  /* ================= SALA DOS AGENTES — cena 3D (19/09/2026) =================
     Pedido explícito do usuário: "tem como nos fazer agentes de ia para rodar
     24h por dia e fazer uma sala 3d onde vai mostar todos agentes
     trabalhando?". A resposta pra "rodar 24h" já existia antes desta tela —
     server.js#start já liga 8 schedulers reais que rodam sozinhos no
     servidor (nunca dependem de ninguém com o navegador aberto). Esta tela
     só dá uma cara visual (3D) pro status desses 8 processos reais, lido de
     GET /api/agentes-3d/status (routes/agentes3d.js) — NENHUM agente/dado
     novo é inventado aqui, e nenhuma lógica de negócio existente é tocada.

     Decisões importantes:
     - Three.js só é carregado (via CDN) na 1ª vez que esta tela abre — nunca
       no carregamento normal do ERP — pra não pesar nas outras telas.
     - "Descarregar" ao sair da tela (ver navigate()) só PAUSA o loop de
       animação e o polling — a cena em si fica guardada na memória. Isso já
       elimina praticamente todo o custo de CPU/GPU/rede em segundo plano, e
       evita destruir/reconstruir o contexto WebGL toda vez que o usuário
       volta pra esta tela (navegadores têm um limite baixo de contextos
       WebGL simultâneos que um destroy/rebuild repetido poderia estourar).
     - Layout fixo (LAYOUT_ORDEM) agrupado por categoria (sincronização,
       financeiro, IA) — mesma ordem sempre, pra o usuário aprender onde
       cada agente fica. */
  (function(){
    const THREE_CDN = 'https://cdn.jsdelivr.net/npm/three@0.128.0/build/three.min.js';
    const ORBIT_CDN = 'https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js';

    // Reescrito em 19/09/2026 (pedido explícito do usuário: "sobre as IA não
    // são aquelas... vou te passar quais são") — só os 5 setores pedidos,
    // cada um com sua própria cor (ver routes/agentes3d.js#categoria) pra
    // ficar fácil de diferenciar visualmente as 5 zonas da sala.
    const LAYOUT_ORDEM = ['promocoes-ia','ads','anuncios-ia','concorrente-ia','sac-ia'];
    const CATEGORIA_TOKEN = { promocoes:'--purple', ads:'--copper', anuncios:'--petrol', concorrente:'--warning', sac:'--teal' };
    const CATEGORIA_LABEL = { promocoes:'Promoções', ads:'Ads', anuncios:'Anúncios', concorrente:'Análise de Concorrente', sac:'SAC' };
    // Mapa estático id->categoria, espelhando routes/agentes3d.js (cada
    // agente já nasce com sua categoria fixa lá). Precisa existir mesmo
    // ANTES do primeiro /status responder, porque construirCena() roda a
    // cor/letreiro de cada zona assim que a cena é montada — usa o dado ao
    // vivo (state.agentesPorId) quando já existir, e cai neste mapa fixo
    // como base honesta (nunca "adivinha" uma categoria nova).
    const CATEGORIA_POR_ID = { 'promocoes-ia':'promocoes', 'ads':'ads', 'anuncios-ia':'anuncios', 'concorrente-ia':'concorrente', 'sac-ia':'sac' };
    function categoriaDoAgente(id){
      return (state.agentesPorId[id] && state.agentesPorId[id].categoria) || CATEGORIA_POR_ID[id] || null;
    }

    const state = {
      shellPronta:false, cenaPronta:false, carregandoScripts:null,
      ativo:false, destruidoEnquantoCarregava:false,
      pollTimer:null, rafId:null,
      agentesPorId:{}, selecionadoId:null,
      // 20/09/2026, pedido explícito do usuário: "quando clicar em cima do
      // agente de ia, quero ver oque ele esta fazendo, oque tenho pra
      // aprovar..., oque ele fez e a melhora que ele teve" — `extraPorId`
      // guarda o resultado de GET /api/ia-agentes/:codigo/detalhe (as outras
      // 3 seções; "o que está fazendo" já existia aqui, vem do status do
      // scheduler em `agentesPorId`). Buscado só quando o personagem é
      // selecionado (nunca a cada poll de 20s — evita bater no banco toda
      // hora com o painel só aberto e parado).
      extraPorId:{},
      three:null, // { renderer, scene, camera, controls, personagens:Map, clock }
    };

    // Um personagem da sala pode representar mais de 1 agente real
    // cadastrado (SAC é um personagem só na sala, mas são 2 agentes
    // separados no catálogo — Mercado Livre e Shopee, ver db/schema.sql).
    const SALA_ID_PARA_AGENTE = {
      'promocoes-ia': ['promocoes'], 'ads': ['ads_performance'],
      'anuncios-ia': ['anuncios_radar'], 'concorrente-ia': ['concorrente'],
      'sac-ia': ['sac_mercado_livre', 'sac_shopee'],
    };

    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function corToken(nomeVar, alt){ const v = getComputedStyle(document.documentElement).getPropertyValue(nomeVar).trim(); return v || alt || '#888888'; }
    function fmtRelativo(iso){
      if(!iso) return 'ainda não rodou';
      const ms = Date.now() - new Date(iso).getTime();
      if(ms < 0) return 'agora há pouco';
      const min = Math.floor(ms/60000);
      if(min < 1) return 'há poucos segundos';
      if(min < 60) return 'há ' + min + ' min';
      const h = Math.floor(min/60);
      if(h < 24) return 'há ' + h + 'h';
      return 'há ' + Math.floor(h/24) + 'd';
    }
    function fmtIntervalo(ms){ if(!ms) return '—'; const min = Math.round(ms/60000); return min < 1 ? Math.round(ms/1000)+'s' : (min+' min'); }

    function carregarScript(src){
      return new Promise((resolve, reject) => {
        const el = document.createElement('script');
        el.src = src; el.async = true;
        el.onload = () => resolve();
        el.onerror = () => reject(new Error('Falha ao carregar ' + src));
        document.head.appendChild(el);
      });
    }
    function garantirThree(){
      if(window.THREE && window.THREE.OrbitControls) return Promise.resolve();
      if(state.carregandoScripts) return state.carregandoScripts;
      state.carregandoScripts = carregarScript(THREE_CDN).then(() => carregarScript(ORBIT_CDN));
      return state.carregandoScripts;
    }

    async function apiStatus(){
      const res = await fetch('/api/agentes-3d/status');
      const body = await res.json().catch(()=>null);
      if(!res.ok) throw new Error((body&&body.error) || 'Não foi possível carregar o status dos agentes.');
      return body.agentes || [];
    }

    // ---- Shell HTML (montada 1 única vez) ----
    function montarShellHTML(){
      const root = document.getElementById('salaAgentesRoot');
      if(!root) return;
      root.innerHTML = '<div class="sa-wrap">'
        + '<div class="sa-canvas-card" id="saCanvasCard">'
          + '<div class="sa-hint">'+svg('eye')+'<span>Arraste para girar · scroll para dar zoom</span></div>'
          + '<div class="sa-legend" id="saLegend"></div>'
          + '<div class="sa-status-overlay" id="saStatusOverlay"><span class="spin"></span><span>Carregando a sala 3D…</span></div>'
        + '</div>'
        + '<div class="sa-side-panel">'
          + '<div class="panel"><div class="panel-head"><div class="panel-title"><span class="panel-title-ic">'+svg('box')+'</span>Agentes</div></div>'
            + '<div id="saListaAgentes"></div>'
          + '</div>'
          + '<div class="panel" id="saPainelDetalhe"><div class="panel-head"><div class="panel-title"><span class="panel-title-ic">'+svg('sparkle')+'</span>Detalhe</div></div>'
            + '<div class="sa-detail-empty" id="saDetalheCorpo">Clique num personagem na sala 3D (ou na lista ao lado) para ver os detalhes.</div>'
          + '</div>'
        + '</div>'
      + '</div>';

      const legend = document.getElementById('saLegend');
      legend.innerHTML = Object.keys(CATEGORIA_LABEL).map(cat =>
        '<div class="sa-legend-row"><span class="sa-legend-dot" style="background:'+corToken(CATEGORIA_TOKEN[cat])+';"></span>'+CATEGORIA_LABEL[cat]+'</div>'
      ).join('');

      state.shellPronta = true;
    }

    function renderListaAgentes(){
      const wrap = document.getElementById('saListaAgentes');
      if(!wrap) return;
      wrap.innerHTML = LAYOUT_ORDEM.map(id => {
        const a = state.agentesPorId[id];
        if(!a) return '';
        const cor = corToken(CATEGORIA_TOKEN[a.categoria]);
        const corStatus = a.emExecucao ? cor : (a.ultimoCicloOk === false ? corToken('--danger') : corToken('--text-faint'));
        return '<div class="conn-row sa-agent-row'+(state.selecionadoId===id?' is-selected':'')+'" data-agente-id="'+id+'">'
          + '<div class="conn-row-left"><span class="conn-ic" style="color:'+corStatus+';">'+svg(a.emExecucao?'activity':'checkCircle')+'</span><div>'
            + '<div class="conn-name">'+escapeHtml(a.nome)+'</div><div class="conn-sub">'+(a.emExecucao?'Trabalhando agora':'Última execução '+fmtRelativo(a.ultimaExecucaoEm))+'</div>'
          + '</div></div>'
        + '</div>';
      }).join('');
      wrap.querySelectorAll('.sa-agent-row').forEach(el => {
        el.addEventListener('click', () => selecionarAgente(el.dataset.agenteId));
      });
    }

    // ---- "O que tenho pra aprovar" / "O que ele já fez" / "A melhora que
    // ele teve" — 20/09/2026, pedido explícito do usuário. "O que está
    // fazendo" (Status/Última execução/Roda a cada) já existia acima, vem
    // do status global do scheduler; estas 3 seções vêm de
    // GET /api/ia-agentes/:codigo/detalhe (por empresa selecionada no
    // filtro do topo) — nunca um número novo calculado aqui no front.
    const SUBSTANTIVO_MELHORA_SALA = { anuncios_radar:'alerta', concorrente:'alerta', sac_mercado_livre:'atendimento', sac_shopee:'atendimento' };

    function combinarDetalhes(detalhes){
      // SAC vira 1 personagem só na sala, mesmo sendo 2 agentes reais — soma
      // pendentes/melhora e junta o histórico dos dois, mais recente primeiro.
      const pendentesTotal = detalhes.reduce((s,d)=> s + (d.pendentes ? d.pendentes.total : 0), 0);
      const temFluxo = detalhes.some((d)=> d.pendentes && d.pendentes.temFluxoDeAprovacao);
      const historico = detalhes.flatMap((d)=> d.historico || []).sort((x,y)=> new Date(y.ts) - new Date(x.ts)).slice(0, 20);
      const melhoras = detalhes.map((d)=> d.melhora).filter(Boolean);
      let melhora = null;
      if(melhoras.length === 1) melhora = melhoras[0];
      else if(melhoras.length > 1 && melhoras[0].tipo === 'resolucao'){
        melhora = {
          tipo: 'resolucao',
          resolvidosUltimos30d: melhoras.reduce((s,m)=> s + m.resolvidosUltimos30d, 0),
          abertosAgora: melhoras.reduce((s,m)=> s + m.abertosAgora, 0),
        };
      }
      return { pendentes: { temFluxoDeAprovacao: temFluxo, total: pendentesTotal }, historico, melhora };
    }

    async function carregarExtra(id){
      const empresaId = window.CerneFiltro && window.CerneFiltro.state ? window.CerneFiltro.state.empresaId : null;
      if(!empresaId){ state.extraPorId[id] = { erro: 'Selecione uma empresa no topo pra ver pendências, histórico e melhora.' }; renderDetalhe(); return; }
      state.extraPorId[id] = { carregando: true };
      renderDetalhe();
      try{
        const codigos = SALA_ID_PARA_AGENTE[id] || [];
        const respostas = await Promise.all(codigos.map((cod)=> fetch('/api/ia-agentes/'+cod+'/detalhe?empresaId='+empresaId).then((r)=> r.json())));
        state.extraPorId[id] = combinarDetalhes(respostas);
      }catch(err){
        state.extraPorId[id] = { erro: 'Não foi possível carregar pendências/histórico/melhora agora.' };
      }
      if(state.selecionadoId === id) renderDetalhe();
    }

    function pendentesSalaHTML(extra, id){
      if(!extra.pendentes.temFluxoDeAprovacao) return '<p class="lead" style="margin:0;">Este agente só observa e recomenda — não existe uma decisão sua pra aprovar aqui.</p>';
      const codigos = SALA_ID_PARA_AGENTE[id] || [];
      const pagina = { promocoes:'agentePromocoes', ads_performance:'agenteAds', concorrente:'concorrente', sac_mercado_livre:'sacMercadoLivre', sac_shopee:'sacShopee' }[codigos[0]];
      const btn = pagina ? '<button class="btn-secondary" type="button" data-ir-pagina="'+pagina+'" style="margin-top:6px;">Ver e decidir</button>' : '';
      return '<p class="lead" style="margin:0 0 4px;"><b>'+extra.pendentes.total+'</b> '+(extra.pendentes.total===1?'sugestão espera':'sugestões esperam')+' sua aprovação.</p>' + btn;
    }
    function historicoSalaHTML(historico){
      if(!historico.length) return '<p class="lead" style="margin:0;">Nenhum evento ainda pra este agente.</p>';
      return '<div class="sa-hist-list">' + historico.slice(0, 8).map((ev)=>
        '<div class="sa-hist-row"><span class="sa-hist-time">'+fmtRelativo(ev.ts)+'</span><span>'+escapeHtml(ev.referencia)+'</span></div>'
      ).join('') + '</div>';
    }
    function melhoraSalaHTML(melhora, id){
      if(!melhora) return '<p class="lead" style="margin:0;">Sem dado de melhora ainda.</p>';
      if(melhora.tipo === 'financeira'){
        if(!melhora.avaliadas) return '<p class="lead" style="margin:0;">Ainda sem decisões avaliadas o suficiente pra medir — a IA reavalia automaticamente alguns dias depois de cada aprovação.</p>';
        const cor = melhora.deltaMedioPct >= 0 ? 'var(--success)' : 'var(--danger)';
        const sinal = melhora.deltaMedioPct >= 0 ? '+' : '';
        return '<p class="lead" style="margin:0;">Margem média <b style="color:'+cor+';">'+sinal+melhora.deltaMedioPct.toLocaleString('pt-BR',{minimumFractionDigits:1,maximumFractionDigits:1})+' p.p.</b> depois da decisão ('+melhora.avaliadas+' avaliada'+(melhora.avaliadas===1?'':'s')+').</p>';
      }
      const subst = SUBSTANTIVO_MELHORA_SALA[(SALA_ID_PARA_AGENTE[id]||[])[0]] || 'item';
      return '<p class="lead" style="margin:0;">Nos últimos 30 dias: <b>'+melhora.resolvidosUltimos30d+'</b> '+subst+(melhora.resolvidosUltimos30d===1?'':'s')+' resolvido'+(melhora.resolvidosUltimos30d===1?'':'s')+', <b>'+melhora.abertosAgora+'</b> ainda aberto'+(melhora.abertosAgora===1?'':'s')+'.</p>';
    }

    function extraHTML(id){
      const extra = state.extraPorId[id];
      if(!extra) return '';
      if(extra.erro) return '<div class="sa-detail-row" style="flex-direction:column;align-items:flex-start;gap:4px;"><p class="lead" style="margin:0;color:var(--warning);">'+escapeHtml(extra.erro)+'</p></div>';
      if(extra.carregando) return '<div class="sa-detail-row"><span class="spin"></span><span>Carregando pendências, histórico e melhora…</span></div>';
      return '<div class="sa-detail-section"><h4>O que você tem pra aprovar</h4>' + pendentesSalaHTML(extra, id) + '</div>'
        + '<div class="sa-detail-section"><h4>O que ele já fez</h4>' + historicoSalaHTML(extra.historico) + '</div>'
        + '<div class="sa-detail-section"><h4>A melhora que ele teve</h4>' + melhoraSalaHTML(extra.melhora, id) + '</div>';
    }

    function renderDetalhe(){
      const corpo = document.getElementById('saDetalheCorpo');
      if(!corpo) return;
      const a = state.selecionadoId ? state.agentesPorId[state.selecionadoId] : null;
      if(!a){ corpo.className = 'sa-detail-empty'; corpo.textContent = 'Clique num personagem na sala 3D (ou na lista ao lado) para ver os detalhes.'; return; }
      corpo.className = '';
      corpo.innerHTML = '<div class="sa-detail-name">'+escapeHtml(a.nome)+'</div>'
        + '<h4>O que está fazendo agora</h4>'
        + '<div class="sa-detail-row"><span>Status</span><span>'+(a.emExecucao ? 'Trabalhando agora' : (a.ativo ? 'Ocioso (aguardando o próximo ciclo)' : 'Parado'))+'</span></div>'
        + '<div class="sa-detail-row"><span>Última execução</span><span>'+fmtRelativo(a.ultimaExecucaoEm)+'</span></div>'
        + '<div class="sa-detail-row"><span>Último ciclo</span><span>'+(a.ultimoCicloOk===null ? '—' : (a.ultimoCicloOk ? 'Concluído sem erro' : 'Terminou com erro'))+'</span></div>'
        + '<div class="sa-detail-row"><span>Roda a cada</span><span>'+fmtIntervalo(a.intervaloMs)+'</span></div>'
        + extraHTML(state.selecionadoId);
      corpo.querySelectorAll('[data-ir-pagina]').forEach((el)=> el.addEventListener('click', ()=> navigate(el.dataset.irPagina)));
    }

    function selecionarAgente(id){
      state.selecionadoId = id;
      renderListaAgentes();
      renderDetalhe();
      document.querySelectorAll('.sa-label').forEach(el => el.classList.toggle('is-selected', el.dataset.agenteId === id));
      if(!state.extraPorId[id]) carregarExtra(id);
    }

    // ---- Cena 3D ----
    // Reescrito em 19/09/2026 (pedido explícito do usuário: "quero que faça
    // uma sala 3D igual ou melhor... 5 zonas bem demarcadas com letreiro por
    // setor, personagens mais trabalhados, iluminação melhor" — dentro do
    // que dá pra fazer só com Three.js, sem depender de nenhum projeto
    // externo de terceiros). 5 setores lado a lado (LAYOUT_ORDEM), cada um
    // com: tapete + parede de fundo na cor do setor, luz spot colorida,
    // divisória entre as zonas, e um personagem mais completo (braços,
    // cadeira, luminária, planta) — tudo geometria simples do Three.js, sem
    // nenhum asset externo pra baixar.
    function construirCena(){
      const THREE = window.THREE;
      const card = document.getElementById('saCanvasCard');
      if(!card) return;
      const largura = card.clientWidth, altura = 560;

      const scene = new THREE.Scene();
      scene.background = new THREE.Color(corToken('--page','#0A0C11'));
      scene.fog = new THREE.Fog(scene.background.getHex(), 24, 46);

      const camera = new THREE.PerspectiveCamera(40, largura/altura, 0.1, 120);
      camera.position.set(0, 11, 17);

      const renderer = new THREE.WebGLRenderer({ antialias:true, alpha:false });
      renderer.setPixelRatio(Math.min(window.devicePixelRatio||1, 2));
      renderer.setSize(largura, altura);
      card.insertBefore(renderer.domElement, card.firstChild);

      const controls = new THREE.OrbitControls(camera, renderer.domElement);
      controls.target.set(0, 0.8, 0);
      controls.enableDamping = true; controls.dampingFactor = 0.08;
      controls.minDistance = 7; controls.maxDistance = 30;
      controls.maxPolarAngle = Math.PI * 0.49;
      controls.update();

      scene.add(new THREE.HemisphereLight(0xffffff, 0x30323d, 0.85));
      const sol = new THREE.DirectionalLight(0xffffff, 0.55);
      sol.position.set(6, 16, 8);
      scene.add(sol);

      const chao = new THREE.Mesh(
        new THREE.PlaneGeometry(30, 18),
        new THREE.MeshStandardMaterial({ color: corToken('--surface-2','#191d26'), roughness:0.95 })
      );
      chao.rotation.x = -Math.PI/2;
      scene.add(chao);

      const grade = new THREE.GridHelper(30, 30, corToken('--border','#262b37'), corToken('--border','#262b37'));
      grade.position.y = 0.01;
      scene.add(grade);

      const paredeMat = new THREE.MeshStandardMaterial({ color: corToken('--surface','#13161d'), roughness:1 });
      const paredeFundo = new THREE.Mesh(new THREE.PlaneGeometry(30, 8), paredeMat);
      paredeFundo.position.set(0, 4, -8);
      scene.add(paredeFundo);

      // 5 zonas lado a lado, uma por setor — mesma ordem de LAYOUT_ORDEM.
      const personagens = new Map();
      const N = LAYOUT_ORDEM.length;
      const espaco = 4.7;
      const largZona = espaco - 0.5;

      // Divisórias entre as zonas (N-1 pilares) — reforça a leitura de "5
      // salas separadas" mesmo com o chão contínuo.
      for(let i=1;i<N;i++){
        const xDiv = (i - (N-1)/2) * espaco;
        const pilar = new THREE.Mesh(
          new THREE.BoxGeometry(0.14, 3, 2.6),
          new THREE.MeshStandardMaterial({ color: corToken('--border','#262b37'), roughness:0.9 })
        );
        pilar.position.set(xDiv, 1.5, 0.2);
        scene.add(pilar);
      }

      LAYOUT_ORDEM.forEach((id, i) => {
        const x = (i - (N-1)/2) * espaco;
        const corHex = corToken(CATEGORIA_TOKEN[categoriaDoAgente(id)], '#888888');
        const cor = new THREE.Color(corHex);

        const grupo = new THREE.Group();
        grupo.position.set(x, 0, 0.2);

        // Tapete da zona, na cor do setor — a "demarcação" pedida.
        const tapete = new THREE.Mesh(
          new THREE.PlaneGeometry(largZona, 3.6),
          new THREE.MeshStandardMaterial({ color: cor, roughness:1, transparent:true, opacity:0.16 })
        );
        tapete.rotation.x = -Math.PI/2;
        tapete.position.set(0, 0.015, 0.3);
        grupo.add(tapete);

        // Painel na parede de fundo, na cor do setor.
        const painel = new THREE.Mesh(
          new THREE.PlaneGeometry(largZona - 0.4, 2.2),
          new THREE.MeshStandardMaterial({ color: cor, roughness:1, transparent:true, opacity:0.22 })
        );
        painel.position.set(0, 2.1, -7.85);
        grupo.add(painel);

        // Luz spot colorida apontando pra mesa — "iluminação melhor" pedida.
        const spot = new THREE.SpotLight(cor, 1.1, 12, Math.PI/5, 0.5, 1.4);
        spot.position.set(0, 5.5, 2.5);
        spot.target.position.set(0, 0.8, 0.3);
        grupo.add(spot);
        grupo.add(spot.target);

        // Mesa
        const mesa = new THREE.Mesh(new THREE.BoxGeometry(1.7, 0.12, 0.9), new THREE.MeshStandardMaterial({ color:'#8a6a45', roughness:0.8 }));
        mesa.position.y = 0.75;
        grupo.add(mesa);
        [[-0.75,-0.35],[0.75,-0.35],[-0.75,0.35],[0.75,0.35]].forEach(([lx,lz]) => {
          const perna = new THREE.Mesh(new THREE.BoxGeometry(0.08, 0.75, 0.08), new THREE.MeshStandardMaterial({ color:'#5c4630' }));
          perna.position.set(lx, 0.375, lz);
          grupo.add(perna);
        });

        // Monitor (acende com a cor do setor quando está trabalhando)
        const monitor = new THREE.Mesh(new THREE.BoxGeometry(0.55, 0.36, 0.05), new THREE.MeshStandardMaterial({ color:'#111318', emissive:new THREE.Color(0x000000) }));
        monitor.position.set(0, 1.12, -0.3);
        grupo.add(monitor);

        // Luminária de mesa — acende junto com o monitor.
        const luminariaBase = new THREE.Mesh(new THREE.CylinderGeometry(0.06,0.09,0.06,10), new THREE.MeshStandardMaterial({ color:'#333' }));
        luminariaBase.position.set(0.65, 0.84, -0.1);
        grupo.add(luminariaBase);
        const luminariaCupula = new THREE.Mesh(new THREE.ConeGeometry(0.12,0.16,12,1,true), new THREE.MeshStandardMaterial({ color:'#333', emissive:new THREE.Color(0x000000), side:THREE.DoubleSide }));
        luminariaCupula.position.set(0.65, 1.1, -0.1);
        luminariaCupula.rotation.x = Math.PI;
        grupo.add(luminariaCupula);

        // Cadeira
        const cadeiraMat = new THREE.MeshStandardMaterial({ color:'#2c313d', roughness:0.8 });
        const assento = new THREE.Mesh(new THREE.BoxGeometry(0.55,0.08,0.55), cadeiraMat);
        assento.position.set(0, 0.55, 0.95);
        grupo.add(assento);
        const encosto = new THREE.Mesh(new THREE.BoxGeometry(0.55,0.55,0.08), cadeiraMat);
        encosto.position.set(0, 0.85, 1.2);
        grupo.add(encosto);

        // Planta — um pouco de vida na zona.
        const vaso = new THREE.Mesh(new THREE.CylinderGeometry(0.14,0.11,0.22,10), new THREE.MeshStandardMaterial({ color:'#6b4a35' }));
        vaso.position.set(-largZona/2 + 0.5, 0.11, -1.1);
        grupo.add(vaso);
        const folhas = new THREE.Mesh(new THREE.SphereGeometry(0.28,10,10), new THREE.MeshStandardMaterial({ color: corToken('--success','#3FB37E'), roughness:1 }));
        folhas.position.set(-largZona/2 + 0.5, 0.5, -1.1);
        grupo.add(folhas);

        // Personagem: corpo + cabeça + 2 braços, cor do setor.
        const corpoMat = new THREE.MeshStandardMaterial({ color: cor });
        const corpo = new THREE.Mesh(new THREE.CylinderGeometry(0.26, 0.32, 0.85, 12), corpoMat);
        corpo.position.set(0, 1.18, 0.55);
        grupo.add(corpo);
        const cabecaMat = new THREE.MeshStandardMaterial({ color:0xe7c9a3 });
        const cabeca = new THREE.Mesh(new THREE.SphereGeometry(0.24, 16, 16), cabecaMat);
        cabeca.position.set(0, 1.78, 0.55);
        grupo.add(cabeca);
        [[-0.32,0],[0.32,0]].forEach(([bx]) => {
          const braco = new THREE.Mesh(new THREE.CylinderGeometry(0.07,0.07,0.55,8), corpoMat);
          braco.position.set(bx, 1.1, 0.65);
          braco.rotation.x = -0.5;
          grupo.add(braco);
        });

        scene.add(grupo);
        personagens.set(id, {
          grupo, corpoMat, monitorMat: monitor.material, luminariaMat: luminariaCupula.material, spotLight: spot,
          baseY: grupo.position.y, faseAnim: Math.random()*Math.PI*2,
        });

        // Rótulo HTML flutuante — dobra como "letreiro" do setor (nome do
        // agente maior/mais destacado que antes) e como status ao vivo.
        const label = document.createElement('div');
        label.className = 'sa-label'; label.dataset.agenteId = id;
        label.innerHTML = '<div class="sa-label-chip"><span class="sa-label-dot" style="background:#888;"></span><span class="sa-label-texto">'+escapeHtml(id)+'</span></div>';
        label.addEventListener('click', () => selecionarAgente(id));
        label.style.pointerEvents = 'auto';
        label.style.cursor = 'pointer';
        card.appendChild(label);
        personagens.get(id).labelEl = label;

        // Placa fixa com o nome do setor, sempre visível acima da parede —
        // separada do chip de status (que já mostra o nome também, mas some
        // se o personagem sair da tela) pra sempre dar pra ler qual é qual.
        const placa = document.createElement('div');
        placa.className = 'sa-letreiro';
        placa.textContent = CATEGORIA_LABEL[categoriaDoAgente(id)] || id;
        placa.style.borderColor = corHex;
        placa.style.color = corHex;
        placa.dataset.agenteId = id;
        card.appendChild(placa);
        personagens.get(id).placaEl = placa;
      });

      state.three = { renderer, scene, camera, controls, personagens, clock: new THREE.Clock(), card };
      state.cenaPronta = true;

      window.addEventListener('resize', onResizeCena);
    }

    function onResizeCena(){
      if(!state.three) return;
      const { renderer, camera, card } = state.three;
      const largura = card.clientWidth;
      camera.aspect = largura/560;
      camera.updateProjectionMatrix();
      renderer.setSize(largura, 560);
    }

    function atualizarPersonagens(){
      const THREE = window.THREE;
      state.three.personagens.forEach((p, id) => {
        const a = state.agentesPorId[id];
        if(!a) return;
        const hex = corToken(CATEGORIA_TOKEN[a.categoria], '#888888');
        const cor = new THREE.Color(hex);
        p.corpoMat.color = cor;
        if(a.emExecucao){
          p.monitorMat.emissive = cor; p.monitorMat.emissiveIntensity = 0.9;
          if(p.luminariaMat){ p.luminariaMat.emissive = cor; p.luminariaMat.emissiveIntensity = 0.8; }
          if(p.spotLight) p.spotLight.intensity = 1.7;
        } else if(a.ultimoCicloOk === false){
          const corErro = new THREE.Color(corToken('--danger','#DC2626'));
          p.monitorMat.emissive = corErro; p.monitorMat.emissiveIntensity = 0.5;
          if(p.luminariaMat){ p.luminariaMat.emissive = corErro; p.luminariaMat.emissiveIntensity = 0.4; }
          if(p.spotLight) p.spotLight.intensity = 0.9;
        } else {
          p.monitorMat.emissive = new THREE.Color(0x000000); p.monitorMat.emissiveIntensity = 0;
          if(p.luminariaMat){ p.luminariaMat.emissive = new THREE.Color(0x000000); p.luminariaMat.emissiveIntensity = 0; }
          if(p.spotLight) p.spotLight.intensity = 1.1;
        }
        const dot = p.labelEl.querySelector('.sa-label-dot');
        const corStatus = a.emExecucao ? hex : (a.ultimoCicloOk === false ? corToken('--danger') : corToken('--text-faint'));
        dot.style.background = corStatus;
        p.labelEl.querySelector('.sa-label-texto').textContent = a.nome.length > 26 ? a.nome.slice(0,24)+'…' : a.nome;
      });
    }

    function loopAnimacao(){
      if(!state.ativo || !state.three) return;
      const { renderer, scene, camera, controls, personagens, clock, card } = state.three;
      const t = clock.getElapsedTime();
      personagens.forEach((p, id) => {
        const a = state.agentesPorId[id];
        const trabalhando = a && a.emExecucao;
        const amp = trabalhando ? 0.05 : 0.02;
        const vel = trabalhando ? 5 : 1.6;
        p.grupo.position.y = p.baseY + Math.sin(t*vel + p.faseAnim) * amp;
        if(trabalhando) p.grupo.rotation.y = Math.sin(t*1.4 + p.faseAnim) * 0.12;
      });
      controls.update();
      renderer.render(scene, camera);

      // Projeta cada rótulo pra posição de tela
      const rect = card.getBoundingClientRect();
      personagens.forEach((p) => {
        const pos = new THREE.Vector3(0, 2.15, 0.55).add(p.grupo.position);
        pos.project(camera);
        const visivel = pos.z < 1;
        p.labelEl.style.display = visivel ? '' : 'none';
        if(visivel){
          p.labelEl.style.left = ((pos.x*0.5+0.5) * rect.width) + 'px';
          p.labelEl.style.top = ((1 - (pos.y*0.5+0.5)) * rect.height) + 'px';
        }

        // Letreiro fixo do setor — projetado acima do painel na parede de
        // fundo (mais alto que o chip de status), sempre visível junto da
        // zona mesmo quando o personagem sai um pouco do enquadramento.
        if(p.placaEl){
          const posPlaca = new THREE.Vector3(0, 3.3, -7.85).add(p.grupo.position);
          posPlaca.project(camera);
          const placaVisivel = posPlaca.z < 1;
          p.placaEl.style.display = placaVisivel ? '' : 'none';
          if(placaVisivel){
            p.placaEl.style.left = ((posPlaca.x*0.5+0.5) * rect.width) + 'px';
            p.placaEl.style.top = ((1 - (posPlaca.y*0.5+0.5)) * rect.height) + 'px';
          }
        }
      });

      state.rafId = requestAnimationFrame(loopAnimacao);
    }

    // ---- Ciclo de dados (polling) ----
    async function atualizarStatus(){
      try{
        const lista = await apiStatus();
        state.agentesPorId = Object.fromEntries(lista.map(a => [a.id, a]));
        // Só remove o overlay quando a cena 3D existe de fato — se ela falhou
        // ao carregar (ex.: CDN bloqueado), load() já escreveu ali a mensagem
        // de erro explicando isso, e essa mensagem não pode ser apagada só
        // porque o status (que é independente da cena) veio com sucesso.
        if(state.cenaPronta){
          const overlay = document.getElementById('saStatusOverlay');
          if(overlay) overlay.remove();
        }
        renderListaAgentes();
        renderDetalhe();
        if(state.cenaPronta) atualizarPersonagens();
      }catch(err){
        const overlay = document.getElementById('saStatusOverlay');
        if(overlay){ overlay.innerHTML = '<span style="color:var(--danger);">'+escapeHtml(err.message||'Não foi possível carregar o status dos agentes.')+'</span>'; }
      }
    }

    async function load(){
      if(!state.shellPronta) montarShellHTML();
      state.ativo = true;

      if(!state.cenaPronta){
        try{
          await garantirThree();
          if(!state.ativo) return; // usuário já saiu da tela enquanto carregava
          construirCena();
        }catch(err){
          const overlay = document.getElementById('saStatusOverlay');
          if(overlay) overlay.innerHTML = '<span style="color:var(--danger);">Não foi possível carregar a sala 3D (biblioteca externa bloqueada?). O status continua disponível na lista ao lado.</span>';
        }
      }

      await atualizarStatus();
      if(!state.ativo) return;

      if(state.cenaPronta && state.rafId === null) loopAnimacao();
      if(state.pollTimer) clearInterval(state.pollTimer);
      state.pollTimer = setInterval(atualizarStatus, 20000);
    }

    function unload(){
      state.ativo = false;
      if(state.pollTimer){ clearInterval(state.pollTimer); state.pollTimer = null; }
      if(state.rafId !== null){ cancelAnimationFrame(state.rafId); state.rafId = null; }
    }

    window.SalaAgentes = { load, unload };
  })();
