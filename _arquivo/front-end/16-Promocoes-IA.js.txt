  /* ================= IA DE PROMOÇÕES — Fase A + Fase B (13/09/2026) =================
     Pedido do usuário: módulo que analisa promoções do Mercado Livre e
     recomenda entrar/sair olhando a MARGEM REAL (nunca só o % de desconto).
     Fase A: 1) o card "Integração Mercado Livre" (leitura vs.
     leitura+escrita, pedido explícito do usuário na seção 0), nunca
     decidido por adivinhação — ver lib/mlPermissoes.js; 2) a configuração
     da margem mínima (seção 3).
     Fase B (pedido explícito do usuário, mesmo dia: "dentro dessa pagina
     quero que mostra todas as promoçoes juntos ao anuncios e a margem de
     cada uma elas, e a cada uma hora a ia tem que busca novas promoções"):
     3) a tabela com toda promoção + anúncio + margem real já calculada
     (lib/promocoesMotor.js) e a classificação da IA (🟢/🟡/🔴/🔵/⚠️); os
     dados vêm sempre de /api/promocoes/analise — já calculados pelo ciclo
     automático que roda a cada 1h no servidor (lib/ia/promocoesScheduler.js,
     nunca depende desta tela estar aberta) — com um botão "Atualizar agora"
     pra forçar uma busca na hora. "Ver detalhes" sempre funciona; Aceitar/
     Remover ficam desabilitados com a mensagem "AÇÃO NÃO DISPONÍVEL" —
     pedido explícito do usuário — enquanto a escrita não estiver liberada
     (ver card de Integração).
     Usa o filtro do topo (window.CerneFiltro) pra empresa, mesmo padrão de
     Margem por Anúncio/Ads/Performance de Anúncios. */
  (function(){
    const state = { statusIntegracao:null, loadingStatus:false, config:null, loadingConfig:false, salvando:false, error:null, analise:null, loadingAnalise:false, atualizandoAgora:false };

    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function money(v){ return v===null||v===undefined ? '—' : 'R$ ' + Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2}); }
    function pctFmt(v){ return v===null||v===undefined ? '—' : (v<0?'− ':'')+Math.abs(Number(v)).toLocaleString('pt-BR',{minimumFractionDigits:1, maximumFractionDigits:1})+'%'; }
    function fmtDataHora(v){
      if(!v) return '—';
      const d = new Date(v);
      return d.toLocaleDateString('pt-BR') + ' ' + d.toLocaleTimeString('pt-BR', { hour:'2-digit', minute:'2-digit' });
    }
    function thumbHTML(url){
      if(!url) return '<span class="anuncio-thumb-placeholder" title="Sem imagem disponível">'+svg('box')+'</span>';
      return '<span style="position:relative;display:inline-block;width:36px;height:36px;flex-shrink:0;">'
        + '<img class="anuncio-thumb" src="'+escapeHtml(url)+'" alt="" loading="lazy" style="position:absolute;inset:0;" onerror="this.style.display=\'none\';this.nextElementSibling.style.display=\'flex\';">'
        + '<span class="anuncio-thumb-placeholder" style="display:none;position:absolute;inset:0;">'+svg('box')+'</span>'
      + '</span>';
    }
    const TIPO_PROMOCAO_LABEL = {
      DEAL:'Oferta (Deal)', MARKETPLACE_CAMPAIGN:'Campanha do marketplace', PRICE_DISCOUNT:'Desconto de preço',
      VOLUME:'Desconto por volume', DOD:'Oferta do dia', LIGHTNING:'Oferta relâmpago', PRE_NEGOTIATED:'Pré-negociada',
      SELLER_CAMPAIGN:'Campanha do vendedor', SMART:'Smart', PRICE_MATCHING:'Igualação de preço',
      UNHEALTHY_STOCK:'Estoque parado', SELLER_COUPON_CAMPAIGN:'Cupom do vendedor',
    };
    function tipoPromocaoLabel(t){ return TIPO_PROMOCAO_LABEL[t] || t || '—'; }
    // Mesma classificação/emoji de lib/promocoesMotor.js#classificar — só
    // repetida aqui pra exibição (o backend já manda o código e o rótulo
    // prontos; o emoji é só cosmético, nunca decide nada sozinho no front).
    const CLASSIFICACAO_COR = {
      entrar:'var(--success)', manter:'var(--success)', oportunidade:'var(--copper)',
      nao_recomendado:'var(--danger)', sair:'var(--danger)',
      risco_margem:'var(--warning)', dados_insuficientes:'var(--warning)',
    };
    const CLASSIFICACAO_EMOJI = {
      entrar:'🟢', manter:'🟡', oportunidade:'🔵', nao_recomendado:'🔴', sair:'🔴',
      risco_margem:'⚠️', dados_insuficientes:'⚠️',
    };
    function badgeClassificacaoHTML(l){
      const cor = CLASSIFICACAO_COR[l.classificacaoCodigo] || 'var(--text-faint)';
      const emoji = CLASSIFICACAO_EMOJI[l.classificacaoCodigo] || '';
      return '<span style="display:inline-flex;align-items:center;gap:5px;font-weight:700;font-size:12.5px;color:'+cor+';" title="'+escapeHtml(l.motivoIncompleto||'')+'">'
        + emoji+' '+escapeHtml(l.classificacaoLabel||'—')
      + '</span>';
    }
    // "Estoque alto" (20/09/2026, pedido explícito do usuário: "sobre, meu
    // estoque daquele produto estiver alto, quero que me avise") — sinal
    // INDEPENDENTE da classificação de margem (ver badgeClassificacaoHTML
    // acima), por isso é um badge separado, nunca substitui o outro. O
    // texto explicativo é remontado aqui a partir dos números que já vêm
    // prontos da API (routes/promocoes.js) — o motivo em texto completo só
    // existe no momento do cálculo (lib/promocoesMotor.js), não é
    // persistido em promocoes_analises.
    function badgeEstoqueAltoHTML(l){
      if(!l.estoqueAlto) return '';
      const titulo = l.coberturaDiasEstoque !== null && l.coberturaDiasEstoque !== undefined
        ? ('No ritmo de vendas dos últimos 90 dias, esse estoque dura ' + Math.round(l.coberturaDiasEstoque) + ' dias.' + (l.estoqueAtual!==null&&l.estoqueAtual!==undefined ? (' Estoque atual: '+l.estoqueAtual+' unidade(s).') : ''))
        : ('Nenhuma venda registrada nos últimos 90 dias, mas ainda há estoque.' + (l.estoqueAtual!==null&&l.estoqueAtual!==undefined ? (' Estoque atual: '+l.estoqueAtual+' unidade(s).') : ''));
      return '<span style="display:inline-flex;align-items:center;gap:4px;font-weight:600;font-size:11.5px;color:var(--copper);margin-top:3px;" title="'+escapeHtml(titulo)+'">📦 Estoque alto'
        + (l.coberturaDiasEstoque!==null&&l.coberturaDiasEstoque!==undefined ? (' ('+Math.round(l.coberturaDiasEstoque)+' dias)') : '')
      + '</span>';
    }
    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+escapeHtml(message)+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 4200);
    }

    async function apiGet(url){
      const res = await fetch(url);
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok) throw new Error((body && (body.error||(body.errors&&JSON.stringify(body.errors)))) || 'Falha na requisição.');
      return body;
    }
    async function apiStatusIntegracao(empresaId){ return apiGet('/api/promocoes/status-integracao?empresaId='+empresaId); }
    async function apiConfig(empresaId){ return apiGet('/api/promocoes/config?empresaId='+empresaId); }
    async function apiSalvarConfig(dados){
      const res = await fetch('/api/promocoes/config', { method:'PUT', headers:{'Content-Type':'application/json'}, body: JSON.stringify(dados) });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok) throw new Error((body && (body.error||(body.errors&&JSON.stringify(body.errors)))) || 'Falha ao salvar configuração.');
      return body;
    }
    async function apiAnalise(empresaId){ return apiGet('/api/promocoes/analise?empresaId='+empresaId); }
    async function apiAnalisarAgora(empresaId){
      const res = await fetch('/api/promocoes/analisar', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ empresaId }) });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok) throw new Error((body && body.error) || 'Falha ao buscar promoções agora.');
      return body;
    }

    async function load(){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId){ state.statusIntegracao=null; state.config=null; state.analise=null; state.loadingStatus=false; state.loadingConfig=false; state.loadingAnalise=false; render(); return; }
      state.loadingStatus = true; state.loadingConfig = true; state.loadingAnalise = true; state.error = null; render();
      try{
        const [statusIntegracao, config, analise] = await Promise.all([
          apiStatusIntegracao(filtro.empresaId),
          apiConfig(filtro.empresaId),
          apiAnalise(filtro.empresaId),
        ]);
        state.statusIntegracao = statusIntegracao;
        state.config = config;
        state.analise = analise;
      }catch(err){ state.error = err.message || 'Não foi possível carregar a IA de Promoções.'; }
      state.loadingStatus = false; state.loadingConfig = false; state.loadingAnalise = false; render();
    }

    window.CerneFiltro.onChange(()=>{
      const page = document.getElementById('page-promocoesIa');
      if(page && page.classList.contains('active')) load();
      else render();
    });

    function cardIntegracaoHTML(){
      const r = state.statusIntegracao && state.statusIntegracao.resumo;
      const ok = !!(r && r.escritaDisponivel);
      return '<div class="table-card" style="padding:20px;margin-bottom:14px;">'
        + '<div class="panel-title" style="margin-bottom:14px;"><span class="panel-title-ic">'+svg('link')+'</span>Integração Mercado Livre</div>'
        + '<div style="display:flex;flex-wrap:wrap;gap:24px;margin-bottom:10px;">'
          + '<div><div class="field-hint" style="margin:0 0 4px;">Status</div><div style="font-weight:600;">'+(ok?'🟢 Leitura e escrita disponíveis':'🟢 Leitura disponível')+'</div></div>'
          + '<div><div class="field-hint" style="margin:0 0 4px;">Alteração de promoções</div><div style="font-weight:600;">'+(ok?'🟢 Escrita disponível':'🔴 Escrita não disponível')+'</div></div>'
        + '</div>'
        + '<div class="field-hint">'+escapeHtml((r && r.mensagem) || 'Seu aplicativo Mercado Livre possui atualmente acesso de leitura. A IA pode analisar e recomendar promoções, mas não pode alterá-las automaticamente.')+'</div>'
      + '</div>';
    }

    // ---- configuração (seção 3/10 do pedido do usuário) ----
    function cardConfigHTML(){
      const c = state.config || {};
      const modos = [
        { v:'somente_analisar', label:'Somente analisar / recomendar' },
        { v:'sugerir_aprovar', label:'Sugerir e aguardar aprovação' },
        { v:'automatico', label:'Automático' },
      ];
      const permiteEscrita = !!(state.statusIntegracao && state.statusIntegracao.resumo && state.statusIntegracao.resumo.escritaDisponivel);
      return '<div class="table-card" style="padding:20px;margin-bottom:14px;">'
        + '<div class="panel-title" style="margin-bottom:14px;"><span class="panel-title-ic">'+svg('percent')+'</span>Configuração da IA de Promoções</div>'
        + '<div class="inline-form" style="margin-top:0;align-items:flex-end;">'
          + '<div class="field narrow"><label>Margem mínima permitida em promoção (%)</label><input type="number" id="promoMargemMinima" min="0" max="100" step="0.1" value="'+(c.margemMinimaPct!==undefined?c.margemMinimaPct:14)+'"></div>'
          + '<div class="field narrow"><label>Margem de conforto p/ promoção com desconto (pontos %)</label><input type="number" id="promoMargemConforto" min="0" max="100" step="0.1" value="'+(c.margemConfortoPct!==undefined?c.margemConfortoPct:0)+'"></div>'
          + '<button class="ghost-btn" id="promoSalvarConfig" type="button" style="cursor:pointer;opacity:1;">'+svg('check')+'<span>Salvar</span></button>'
        + '</div>'
        + '<div class="field-hint" style="margin-top:6px;">Quando uma promoção deixar a margem estimada abaixo deste número, a IA vai marcar como <strong>🔴 NÃO RECOMENDADA</strong>.</div>'
        + '<div class="field-hint" style="margin-top:4px;">Margem de conforto (15/09/2026): pontos percentuais exigidos ACIMA da margem mínima, mas só pra promoções que têm desconto de verdade (preço abaixo do normal). Uma promoção no preço normal (sem desconto, margem igual à sua margem normal) continua exigindo só a margem mínima. Deixe em 0 para desligar — a IA volta a exigir só a margem mínima em qualquer caso.</div>'
        + '<div style="margin-top:18px;">'
          + '<div class="field-hint" style="margin:0 0 8px;">Modo da IA</div>'
          + modos.map(m => {
              const disabled = m.v !== 'somente_analisar' && !permiteEscrita;
              return '<label style="display:flex;align-items:center;gap:8px;margin-bottom:6px;'+(disabled?'opacity:.5;':'cursor:pointer;')+'" title="'+(disabled?'É necessário habilitar permissão de escrita na integração Mercado Livre.':'')+'">'
                + '<input type="radio" name="promoModoIa" value="'+m.v+'" '+(c.modoIa===m.v?'checked':'')+' '+(disabled?'disabled':'')+'>'
                + '<span>'+escapeHtml(m.label)+'</span>'
              + '</label>';
            }).join('')
        + '</div>'
      + '</div>';
    }

    // ---- tabela: todas as promoções junto com o anúncio e a margem real
    // de cada uma (pedido explícito do usuário, 13/09/2026) — lê sempre de
    // /api/promocoes/analise (já calculado pelo ciclo automático de hora em
    // hora, lib/ia/promocoesCiclo.js; nunca recalcula ao vivo aqui). ----
    function anuncioCellHTML(l){
      return '<td class="produto-cell"><a href="javascript:void(0)" class="promo-anuncio-link" data-ml-item="'+escapeHtml(l.mlItemId||'')+'" style="text-decoration:none;color:inherit;"><div class="anuncio-cell-flex">'
        + thumbHTML(l.imagemUrl)
        + '<div class="cell-stack">'
          + '<span class="prod-title" title="'+escapeHtml(l.titulo||'—')+'" style="text-decoration:underline dotted;">'+escapeHtml(l.titulo||'(sem título)')+'</span>'
          + '<span class="cell-stack-sub mono">'+escapeHtml(l.sku||'sem SKU')+'</span>'
        + '</div></div></a></td>';
    }

      const a = state.analise;

          + (state.atualizandoAgora ? '<span class="spin" style="width:13px;height:13px;"></span>' : svg('activity'))

        + '</button>'
      + '</div>';
    }

      return '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
        + '<th>Anúncio / SKU</th><th>Loja</th><th>Promoção</th><th>Status</th><th>Preço normal</th><th>Preço promo.</th>'

          + '<td class="mono">'+money(l.precoPromo)+'</td>'

          + '<td class="mono" style="'+(l.margemRealPct!==null?(l.margemRealPct>=0?'color:var(--success);':'color:var(--danger);'):'')+'">'+(l.margemIncompleta?'<span style="color:var(--text-faint);font-style:italic;" title="'+escapeHtml(l.motivoIncompleto||'')+'">dados insuficientes</span>':pctFmt(l.margemRealPct))+'</td>'

          + '<td style="white-space:nowrap;">'

            + ' <button class="row-ic-btn" type="button" disabled title="AÇÃO NÃO DISPONÍVEL: esta ação exige permissão de escrita na API do Mercado Livre. Atualmente sua aplicação possui apenas permissão de leitura." style="opacity:.4;cursor:not-allowed;">'+svg('checkCircle')+'</button>'
            + ' <button class="row-ic-btn" type="button" disabled title="AÇÃO NÃO DISPONÍVEL: esta ação exige permissão de escrita na API do Mercado Livre. Atualmente sua aplicação possui apenas permissão de leitura." style="opacity:.4;cursor:not-allowed;">'+svg('x')+'</button>'

    function render(){
      const root = document.getElementById('promocoesIaRoot');
      if(!root) return;
      const filtro = window.CerneFiltro.state;

      if(!filtro.loadingEmpresas && !filtro.empresas.length){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
          + '<h2>Nenhuma empresa ativa</h2>'
          + '<p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> antes de usar a IA de Promoções.</p></div>';
        return;
      }
      if(!filtro.empresaId){ root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('sparkle')+'</div><h2>Selecione uma empresa</h2></div>'; return; }

      if((state.loadingStatus || state.loadingConfig) && !state.statusIntegracao){
        root.innerHTML = '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando IA de Promoções…</span></div></div>';
        return;
      }
      if(state.error){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
          + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p>'
          + '<button class="ghost-btn" id="promoRetry" style="cursor:pointer; opacity:1;">'+svg('activity')+'<span>Tentar novamente</span></button></div>';
        root.querySelector('#promoRetry').addEventListener('click', load);
        return;
      }

      let html = cardIntegracaoHTML() + cardConfigHTML() + cardResumoAnaliseHTML();

      if(state.loadingAnalise && !state.analise){
        html += '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Buscando promoções…</span></div></div>';
      } else if(state.analise && !state.analise.linhas.length){
        html += '<div class="empty-panel"><div class="empty-icon">'+svg('sparkle')+'</div>'
          + '<h2>Nenhuma promoção encontrada ainda</h2>'
          + '<p class="lead">A busca automática roda a cada 1 hora. Você também pode clicar em <strong>Atualizar agora</strong> para forçar uma busca neste instante.</p></div>';
      } else if(state.analise){
        html += tabelaPromocoesHTML(state.analise.linhas);
      }

      root.innerHTML = html;

      const btnAtualizarAgora = document.getElementById('promoAtualizarAgora');
      if(btnAtualizarAgora) btnAtualizarAgora.addEventListener('click', async ()=>{
        state.atualizandoAgora = true; render();
        try{
          await apiAnalisarAgora(filtro.empresaId);
          state.analise = await apiAnalise(filtro.empresaId);
          toast('Promoções atualizadas.');
        }catch(err){ toast(err.message || 'Não foi possível buscar promoções agora.', 'error'); }
        state.atualizandoAgora = false; render();
      });

      root.querySelectorAll('.promo-anuncio-link, .promo-ver-detalhes').forEach(el=>{
        el.addEventListener('click', ()=>{
          const mlItemId = el.dataset.mlItem;
          const l = (state.analise && state.analise.linhas || []).find(x=>x.mlItemId===mlItemId);
          if(l) abrirDetalhePromocao(l);
        });
      });

      const btnSalvar = document.getElementById('promoSalvarConfig');
      if(btnSalvar) btnSalvar.addEventListener('click', async ()=>{
        const margemMinimaPct = Number(document.getElementById('promoMargemMinima').value);
        const margemConfortoPct = Number(document.getElementById('promoMargemConforto').value);
        if(!Number.isFinite(margemMinimaPct) || margemMinimaPct < 0 || margemMinimaPct > 100){
          toast('Informe uma margem mínima entre 0 e 100.', 'error'); return;
        }
        if(!Number.isFinite(margemConfortoPct) || margemConfortoPct < 0 || margemConfortoPct > 100){
          toast('Informe uma margem de conforto entre 0 e 100.', 'error'); return;
        }
        btnSalvar.disabled = true;
        try{
          state.config = await apiSalvarConfig({ empresaId: filtro.empresaId, margemMinimaPct, margemConfortoPct });
          toast('Configuração salva.');
          render();
        }catch(err){ toast(err.message, 'error'); }
        finally{ if(btnSalvar) btnSalvar.disabled = false; }
      });
      root.querySelectorAll('input[name="promoModoIa"]').forEach(r=> r.addEventListener('change', async (e)=>{
        try{
          state.config = await apiSalvarConfig({ empresaId: filtro.empresaId, modoIa: e.target.value });
          toast('Configuração salva.');
        }catch(err){ toast(err.message, 'error'); }
        render();
      }));
    }

    window.PromocoesIA = { load };
  })();
