  /* ================= FORNECEDORES — cadastro por empresa =================
     Razão social/nome, nome fantasia, CNPJ ou CPF, telefone, e-mail,
     observação e status. Estrutura pronta para futuramente relacionar
     fornecedor a produtos e compras — a relação em si ainda não existe. */
  (function(){
    const state = { empresas:[], empresaId:null, loadingEmpresas:false, items:[], loading:false, error:null, status:'todos', search:'' };

    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+message+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 3200);
    }
    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function fmtDate(iso){ try{ return new Date(iso).toLocaleDateString('pt-BR'); }catch(e){ return '—'; } }

    function maskDocumento(v){
      const d = v.replace(/\D/g,'').slice(0,14);
      if(d.length <= 11){
        let out = d;
        if(d.length > 9) out = d.replace(/(\d{3})(\d{3})(\d{3})(\d{0,2})/, '$1.$2.$3-$4');
        else if(d.length > 6) out = d.replace(/(\d{3})(\d{3})(\d{0,3})/, '$1.$2.$3');
        else if(d.length > 3) out = d.replace(/(\d{3})(\d{0,3})/, '$1.$2');
        return out;
      }
      return d.replace(/(\d{2})(\d{3})(\d{3})(\d{4})(\d{0,2})/, '$1.$2.$3/$4-$5');
    }

    async function apiEmpresasAtivas(){
      const res = await fetch('/api/empresas?status=ativas');
      return ((await res.json()).empresas) || [];
    }
    async function api(path, opts){
      const res = await fetch('/api/fornecedores'+path, { headers:{'Content-Type':'application/json'}, ...opts });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok){ const err = new Error((body&&body.error)||'Falha na requisição.'); err.errors = body&&body.errors; throw err; }
      return body;
    }

    async function loadItens(){
      state.loading = true; state.error = null; render();
      try{
        const params = new URLSearchParams({ empresaId: state.empresaId });
        if(state.status !== 'todos') params.set('status', state.status);
        if(state.search.trim()) params.set('search', state.search.trim());
        const data = await api('?'+params.toString(), { method:'GET' });
        state.items = data.fornecedores;
      }catch(err){ state.error = err.message || 'Não foi possível carregar os fornecedores.'; }
      finally{ state.loading = false; render(); }
    }

    async function load(){
      state.loadingEmpresas = true; render();
      try{
        state.empresas = await apiEmpresasAtivas();
        if(!state.empresaId || !state.empresas.find(e=>e.id===state.empresaId)){
          state.empresaId = state.empresas.length ? state.empresas[0].id : null;
        }
      }catch(e){ state.empresas = []; }
      state.loadingEmpresas = false;
      if(state.empresaId) await loadItens(); else render();
    }

    function render(){
      const root = document.getElementById('fornecedoresRoot');
      if(!root) return;

      if(!state.loadingEmpresas && !state.empresas.length){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
          + '<h2>Nenhuma empresa ativa</h2>'
          + '<p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> antes de cadastrar fornecedores.</p></div>';
        return;
      }

      let html = '<div class="toolbar">'
        + '<div class="empresa-picker" style="margin-bottom:0;"><select id="forEmpresaSelect">'
          + state.empresas.map(e=>'<option value="'+e.id+'" '+(e.id===state.empresaId?'selected':'')+'>'+escapeHtml(e.nomeFantasia||e.razaoSocial)+'</option>').join('')
        + '</select></div>'
        + '<div class="seg" id="forSeg">'
          + [['todos','Todos'],['ativos','Ativos'],['inativos','Inativos']].map(function(par){ return '<button data-status="'+par[0]+'" class="'+(state.status===par[0]?'active':'')+'">'+par[1]+'</button>'; }).join('')
        + '</div>'
        + '<div class="search-box">'+svg('search')+'<input type="text" id="forSearch" placeholder="Buscar por razão social, fantasia ou CNPJ/CPF" value="'+state.search.replace(/"/g,'&quot;')+'"></div>'
      + '</div>';

      if(state.loading){
        html += '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando fornecedores…</span></div></div>';
      } else if(state.error){
        html += '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
          + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p>'
          + '<button class="ghost-btn" id="forRetry" style="cursor:pointer; opacity:1;">'+svg('activity')+'<span>Tentar novamente</span></button></div>';
      } else if(!state.items.length){
        const filtrado = !!(state.search || state.status!=='todos');
        html += '<div class="empty-panel"><div class="empty-icon">'+svg('truck')+'</div>'
          + '<h2>Nenhum fornecedor encontrado</h2>'
          + '<p class="lead">'+(filtrado ? 'Nenhum fornecedor corresponde ao filtro atual.' : 'Cadastre o primeiro fornecedor (razão social e CNPJ/CPF) desta empresa.')+'</p>'
          + (filtrado ? '' : '<button class="primary-btn enabled" id="forEmptyNew" type="button">'+svg('plus')+'<span>Cadastrar fornecedor</span></button>')
        + '</div>';
      } else {
        html += '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>Fornecedor</th><th>CNPJ/CPF</th><th>Telefone</th><th>E-mail</th><th>Status</th><th></th>'
          + '</tr></thead><tbody>'
          + state.items.map(f => '<tr class="'+(f.ativo?'':'inactive')+'">'
            + '<td><div class="company-name">'+escapeHtml(f.razaoSocial)+'</div>'+(f.nomeFantasia?'<div class="company-fantasy">'+escapeHtml(f.nomeFantasia)+'</div>':'')+'</td>'
            + '<td class="mono">'+escapeHtml(f.documentoFormatado)+'</td>'
            + '<td class="mono">'+escapeHtml(f.telefone||'—')+'</td>'
            + '<td class="mono">'+escapeHtml(f.email||'—')+'</td>'
            + '<td>'+(f.ativo ? '<span class="badge-status on"><span class="dot"></span>Ativo</span>' : '<span class="badge-status off"><span class="dot"></span>Inativo</span>')+'</td>'
            + '<td><div class="row-actions">'
              + '<button class="row-ic-btn" data-edit="'+f.id+'" title="Editar" type="button">'+svg('edit')+'</button>'
              + '<button class="row-ic-btn '+(f.ativo?'danger':'success')+'" data-toggle="'+f.id+'" data-next="'+(!f.ativo)+'" title="'+(f.ativo?'Desativar':'Ativar')+'" type="button">'+svg('power')+'</button>'
            + '</div></td>'
          + '</tr>').join('')
          + '</tbody></table></div></div>';
      }

      root.innerHTML = html;

      const sel = document.getElementById('forEmpresaSelect');
      if(sel) sel.addEventListener('change', ()=>{ state.empresaId = Number(sel.value); loadItens(); });
      const seg = document.getElementById('forSeg');
      if(seg) seg.querySelectorAll('button').forEach(b=> b.addEventListener('click', ()=>{ state.status = b.dataset.status; loadItens(); }));
      const search = document.getElementById('forSearch');
      if(search) search.addEventListener('input', ()=>{
        state.search = search.value;
        clearTimeout(search._t);
        search._t = setTimeout(loadItens, 300);
      });
      const retry = document.getElementById('forRetry');
      if(retry) retry.addEventListener('click', loadItens);
      const emptyNew = document.getElementById('forEmptyNew');
      if(emptyNew) emptyNew.addEventListener('click', ()=> openModal());
      root.querySelectorAll('[data-edit]').forEach(b=> b.addEventListener('click', ()=>{
        const f = state.items.find(x=> String(x.id)===b.dataset.edit);
        openModal(f);
      }));
      root.querySelectorAll('[data-toggle]').forEach(b=> b.addEventListener('click', async ()=>{
        const id = b.dataset.toggle, next = b.dataset.next === 'true';
        b.disabled = true;
        try{
          await api('/'+id+'/status', { method:'PATCH', body: JSON.stringify({ ativo: next }) });
          toast(next ? 'Fornecedor ativado.' : 'Fornecedor desativado.');
          loadItens();
        }catch(err){ toast(err.message, 'error'); b.disabled = false; }
      }));
    }

    function openModal(fornecedor){
      const isEdit = !!fornecedor;
      const modalRoot = document.getElementById('modalRoot');
      modalRoot.innerHTML = '<div class="modal-overlay" id="modalOverlay">'
        + '<div class="modal-card">'
          + '<div class="modal-head"><div class="modal-title">'+(isEdit?'Editar fornecedor':'Novo fornecedor')+'</div>'
            + '<button class="row-ic-btn" id="modalClose" type="button">'+svg('x')+'</button></div>'
          + '<div class="modal-sub">'+(isEdit?'Atualize os dados do fornecedor.':'Preencha os dados para cadastrar um novo fornecedor desta empresa.')+'</div>'
          + '<form id="forForm">'
            + '<div class="field"><label>Razão social / Nome</label><input type="text" id="fRazao" placeholder="Razão social ou nome completo" value="'+(isEdit?escapeHtml(fornecedor.razaoSocial):'')+'" maxlength="200">'
              + '<div class="field-error" id="eRazao" style="display:none;"></div></div>'
            + '<div class="field"><label>Nome fantasia <span class="opt">(opcional)</span></label><input type="text" id="fFantasia" placeholder="Nome usado no dia a dia" value="'+(isEdit?escapeHtml(fornecedor.nomeFantasia||''):'')+'" maxlength="200"></div>'
            + '<div class="field"><label>CNPJ ou CPF</label><input type="text" inputmode="numeric" id="fDocumento" placeholder="00.000.000/0000-00 ou 000.000.000-00" value="'+(isEdit?maskDocumento(fornecedor.documento):'')+'" maxlength="18">'
              + '<div class="field-error" id="eDocumento" style="display:none;"></div></div>'
            + '<div class="field"><label>Telefone <span class="opt">(opcional)</span></label><input type="text" id="fTelefone" placeholder="(00) 00000-0000" value="'+(isEdit?escapeHtml(fornecedor.telefone||''):'')+'" maxlength="20"></div>'
            + '<div class="field"><label>E-mail <span class="opt">(opcional)</span></label><input type="text" id="fEmail" placeholder="contato@fornecedor.com" value="'+(isEdit?escapeHtml(fornecedor.email||''):'')+'" maxlength="255">'
              + '<div class="field-error" id="eEmail" style="display:none;"></div></div>'
            + '<div class="field"><label>Observação <span class="opt">(opcional)</span></label><input type="text" id="fObservacao" placeholder="Condições, contato, observações gerais" value="'+(isEdit?escapeHtml(fornecedor.observacao||''):'')+'"></div>'
            + (isEdit ? '<div class="toggle-row"><div><div class="toggle-row-label">Fornecedor ativo</div><div class="toggle-row-sub">Fornecedores inativos ficam ocultos nas seleções do sistema.</div></div>'
              + '<button type="button" class="toggle '+(fornecedor.ativo?'on':'')+'" id="fAtivo" data-on="'+(fornecedor.ativo?'true':'false')+'"></button></div>' : '')
            + '<div class="modal-actions">'
              + '<button type="button" class="btn-secondary" id="modalCancel">Cancelar</button>'
              + '<button type="submit" class="btn-primary" id="modalSave">'+svg('check')+'<span>'+(isEdit?'Salvar alterações':'Cadastrar fornecedor')+'</span></button>'
            + '</div>'
          + '</form>'
        + '</div>'
      + '</div>';

      const overlay = document.getElementById('modalOverlay');
      const close = ()=> modalRoot.innerHTML = '';
      overlay.addEventListener('click', (e)=>{ if(e.target === overlay) close(); });
      document.getElementById('modalClose')?.addEventListener('click', close);
      document.getElementById('modalCancel')?.addEventListener('click', close);

      const docInput = document.getElementById('fDocumento');
      docInput.addEventListener('input', ()=>{ docInput.value = maskDocumento(docInput.value); });
      document.getElementById('fRazao').focus();

      const toggleBtn = document.getElementById('fAtivo');
      if(toggleBtn) toggleBtn.addEventListener('click', ()=>{
        const on = toggleBtn.dataset.on !== 'true';
        toggleBtn.dataset.on = String(on);
        toggleBtn.classList.toggle('on', on);
      });

      document.getElementById('forForm')?.addEventListener('submit', async (e)=>{
        e.preventDefault();
        ['eRazao','eDocumento','eEmail'].forEach(id=>{ const el=document.getElementById(id); el.style.display='none'; el.textContent=''; });
        ['fRazao','fDocumento','fEmail'].forEach(id=> document.getElementById(id).classList.remove('invalid'));

        const payload = {
          empresaId: state.empresaId,
          razaoSocial: document.getElementById('fRazao').value,
          nomeFantasia: document.getElementById('fFantasia').value,
          documento: document.getElementById('fDocumento').value,
          telefone: document.getElementById('fTelefone').value,
          email: document.getElementById('fEmail').value,
          observacao: document.getElementById('fObservacao').value,
        };
        if(isEdit && toggleBtn) payload.ativo = toggleBtn.dataset.on === 'true';

        const saveBtn = document.getElementById('modalSave');
        saveBtn.disabled = true;
        try{
          if(isEdit) await api('/'+fornecedor.id, { method:'PUT', body: JSON.stringify(payload) });
          else await api('', { method:'POST', body: JSON.stringify(payload) });
          close();
          toast(isEdit ? 'Fornecedor atualizado.' : 'Fornecedor cadastrado.');
          loadItens();
        }catch(err){
          saveBtn.disabled = false;
          if(err.errors){
            const fieldMap = { razaoSocial:'Razao', documento:'Documento', email:'Email' };
            Object.entries(err.errors).forEach(([k,msg])=>{
              const suf = fieldMap[k]; if(!suf) return;
              const el=document.getElementById('e'+suf); if(el){ el.textContent=msg; el.style.display='block'; }
              const input=document.getElementById('f'+suf); if(input) input.classList.add('invalid');
            });
          } else {
            toast(err.message, 'error');
          }
        }
      });
    }

    document.getElementById('btnNewFornecedor')?.addEventListener('click', ()=> openModal());

    window.Fornecedores = { load };
  })();
