  /* ================= CALCULADORA DE VENDAS (Venda de Balcão) =================
     Ativado em 22/09/2026 — pedido explícito do usuário: "quero colocar
     agora uma calculadora de vendas pra eu vende[r] para o cliente final
     que vem até a minha empresa... porque todas as vendas que eu faço pro
     fora eu nao coloco no faturamento sei que isso é errado mas quero
     arrumar". Uma venda presencial (sempre à vista), sempre de produto já
     cadastrado (nunca digitado solto — reaproveita GET /api/produtos), que
     passa a contar em TODO o sistema (Visão Geral/DRE/Relatórios/
     Faturamento — ver o novo canal 'balcao' em lib/relatorioVendas.js) e
     baixa o estoque "fora do Full" automaticamente (ver
     lib/estoqueFisico.js). Usa o filtro do HEADER (window.CerneFiltro) pra
     empresa e período do histórico, igual Faturamento/Notas Fiscais/Pedidos
     — o carrinho de uma venda nova sempre usa a empresa selecionada ali. */
  (function(){
    const FORMA_PAGAMENTO_OPCOES = [
      ['dinheiro','Dinheiro'], ['pix','Pix'], ['debito','Débito'], ['credito','Crédito'], ['outro','Outro'],
    ];
    const FORMA_PAGAMENTO_LABEL = Object.fromEntries(FORMA_PAGAMENTO_OPCOES);
    const state = { loading:false, error:null, vendas:[], search:'' };

    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+message+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 3200);
    }
    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function round2(n){ return Math.round(n*100)/100; }
    function money(v){ return v===null||v===undefined ? '—' : 'R$ ' + Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2}); }
    function fmtDateTime(iso){ if(!iso) return '—'; try{ const d=new Date(iso); return d.toLocaleDateString('pt-BR')+' '+d.toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'}); }catch(e){ return '—'; } }

    async function api(path, opts){
      const res = await fetch('/api/vendas-balcao'+path, { headers:{'Content-Type':'application/json'}, ...opts });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok){ const err = new Error((body&&(body.error||(body.errors&&body.errors.geral)))||'Falha na requisição.'); err.errors = body&&body.errors; throw err; }
      return body;
    }
    async function apiProdutosAtivos(empresaId){
      const res = await fetch('/api/produtos?'+new URLSearchParams({ empresaId, status:'ativos' }).toString());
      const body = await res.json().catch(()=>({}));
      if(!res.ok) throw new Error((body&&body.error) || 'Não foi possível carregar os produtos.');
      return body.produtos || [];
    }

    async function loadVendas(){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId){ state.vendas=[]; state.loading=false; render(); return; }
      state.loading = true; state.error = null; render();
      try{
        const params = new URLSearchParams({ empresaId: filtro.empresaId, periodo: filtro.periodo });
        if(filtro.periodo === 'personalizado'){ if(filtro.desdePersonalizado) params.set('desde', filtro.desdePersonalizado); if(filtro.atePersonalizado) params.set('ate', filtro.atePersonalizado); }
        if(state.search.trim()) params.set('search', state.search.trim());
        const data = await api('?'+params.toString(), { method:'GET' });
        state.vendas = data.vendas;
      }catch(err){ state.error = err.message || 'Não foi possível carregar as vendas de balcão.'; }
      finally{ state.loading = false; render(); }
    }

    // ---- Calculadora de PREÇO (02/10/2026, pedido explícito do usuário:
    // "quero que dê pra eu digitar qual kit eu vou vender e quero que dê
    // pra colocar a margem que vou vender... a calculadora dá o preço que
    // deve ser passado para o cliente"). Mesma definição de margem usada no
    // resto do sistema (lib/relatorioVendas.js): margem = venda − taxas −
    // frete − imposto − custo do produto, e a margem % é sobre o preço de
    // venda. Aqui o caminho é o inverso: dado o custo (cadastrado em
    // Produtos), o imposto da empresa e a margem % desejada, acha o preço:
    //   preço = (custo×qtd + taxa fixa + frete) ÷ (1 − imposto% − taxa% − margem%)
    // É só uma conta de tela — nunca grava nada. O custo vem sempre do
    // cadastro de Produtos (nunca digitado solto); taxa do marketplace e
    // frete ficam por conta do usuário (começam em 0), pra nunca inventar
    // uma tarifa que o sistema não sabe. ----
    const calc = { empresaId:null, loading:false, produtos:[], impostoPct:0, busca:'', qtd:'1', margem:'20', taxaPct:'0', taxaFixa:'0', frete:'0', impostoEditado:null };

    function numBRsimple(v){ const s = String(v==null?'':v).trim(); if(!s) return 0; const n = Number(s.indexOf(',')>=0 ? s.replace(/\./g,'').replace(',','.') : s); return Number.isFinite(n) ? n : NaN; }
    function pctTxt(v){ return Number(v).toLocaleString('pt-BR',{minimumFractionDigits:1, maximumFractionDigits:2})+'%'; }

    function calcAcharProduto(txt){
      const t = String(txt||'').trim().toLowerCase();
      if(!t) return null;
      let p = calc.produtos.find(x=> (x.sku+' — '+x.nome).toLowerCase()===t);
      if(!p) p = calc.produtos.find(x=> String(x.sku).trim().toLowerCase()===t);
      if(!p){ const porNome = calc.produtos.filter(x=> String(x.nome).trim().toLowerCase()===t); if(porNome.length===1) p = porNome[0]; }
      return p || null;
    }

    function calcResultadoHTML(){
      const prod = calcAcharProduto(calc.busca);
      if(!calc.busca.trim()) return '<div class="field-hint">Digite o SKU ou nome do kit acima (aparecem sugestões do seu cadastro de Produtos), informe a margem que quer ter e o preço aparece aqui.</div>';
      if(!prod) return '<div class="field-hint" style="color:var(--warning,#b45309);">Não achei esse kit no cadastro de Produtos ativos. Escolha uma das sugestões da lista (ou cadastre o produto em Cadastros → Produtos).</div>';
      if(!(prod.custo>0)) return '<div class="field-hint" style="color:var(--danger);">Esse produto está com custo zerado no cadastro — atualize o custo em Cadastros → Produtos (ou por planilha) pra calcular o preço.</div>';
      const qtd = numBRsimple(calc.qtd), margem = numBRsimple(calc.margem), taxa = numBRsimple(calc.taxaPct), fixa = numBRsimple(calc.taxaFixa), frete = numBRsimple(calc.frete);
      const imposto = calc.impostoEditado!==null ? numBRsimple(calc.impostoEditado) : calc.impostoPct;
      if([qtd,margem,taxa,fixa,frete,imposto].some(n=>Number.isNaN(n)) || !(qtd>0)) return '<div class="field-hint" style="color:var(--danger);">Confira os números digitados — algum campo está inválido.</div>';
      const denom = 1 - (imposto+taxa+margem)/100;
      if(!(denom>0)) return '<div class="field-hint" style="color:var(--danger);">Impossível: imposto ('+pctTxt(imposto)+') + taxa ('+pctTxt(taxa)+') + margem ('+pctTxt(margem)+') passam de 100% do preço. Diminua a margem.</div>';
      const custoTotal = round2(prod.custo*qtd);
      const preco = (custoTotal + fixa + frete) / denom;
      const precoR = Math.ceil(preco*100)/100; // arredonda pra cima — nunca ficar abaixo da margem pedida
      const valImposto = round2(precoR*imposto/100), valTaxa = round2(precoR*taxa/100);
      const lucro = round2(precoR - custoTotal - fixa - frete - valImposto - valTaxa);
      const margemReal = precoR>0 ? (lucro/precoR)*100 : 0;
      return '<div class="kpi-hero" style="--tint:var(--success);--tint-soft:var(--success-soft);max-width:420px;margin-bottom:10px;">'
          + '<div class="kpi-hero-label">Preço a passar pro cliente'+(qtd!==1?' ('+qtd+' un.)':'')+'</div>'
          + '<div class="kpi-hero-value">'+money(precoR)+'</div>'
          + '<div class="kpi-hero-foot"><span class="kpi-hero-note">'+(qtd>1 ? money(round2(precoR/qtd))+' por unidade · ' : '')+'Lucro de '+money(lucro)+' ('+pctTxt(margemReal)+')</span></div>'
        + '</div>'
        + '<div style="max-width:420px;">'
          + '<div class="bd-row"><span class="bd-label">Preço de venda</span><span class="bd-value">'+money(precoR)+'</span></div>'
          + '<div class="bd-row neg"><span class="bd-label">(−) Custo do produto ('+escapeHtml(prod.sku)+' × '+qtd+')</span><span class="bd-value">− '+money(custoTotal)+'</span></div>'
          + '<div class="bd-row neg"><span class="bd-label">(−) Imposto ('+pctTxt(imposto)+')</span><span class="bd-value">− '+money(valImposto)+'</span></div>'
          + (taxa>0||fixa>0 ? '<div class="bd-row neg"><span class="bd-label">(−) Taxas do marketplace</span><span class="bd-value">− '+money(round2(valTaxa+fixa))+'</span></div>' : '')
          + (frete>0 ? '<div class="bd-row neg"><span class="bd-label">(−) Frete pago por você</span><span class="bd-value">− '+money(frete)+'</span></div>' : '')
          + '<div class="bd-row total"><span class="bd-label">= Lucro (margem de contribuição)</span><span class="bd-value positive">'+money(lucro)+'</span></div>'
        + '</div>';
    }

    function calcPrecoHTML(){
      const impostoVal = calc.impostoEditado!==null ? calc.impostoEditado : String(calc.impostoPct).replace('.',',');
      return '<div class="panel" style="margin-bottom:16px;">'
        + '<div class="panel-head"><div class="panel-title"><span class="panel-title-ic">'+svg('wallet')+'</span>Calculadora de preço</div></div>'

        + '<div class="inline-form" style="margin-top:0;">'
          + '<div class="field" style="min-width:260px;"><label>Kit / produto <span class="opt">(SKU ou nome)</span></label><input type="text" id="calcBusca" list="calcProdutosList" autocomplete="off" placeholder="Digite o SKU ou o nome do kit" value="'+escapeHtml(calc.busca)+'">'
            + '<datalist id="calcProdutosList">'+calc.produtos.map(p=>'<option value="'+escapeHtml(p.sku+' — '+p.nome)+'"></option>').join('')+'</datalist></div>'
          + '<div class="field narrow" style="max-width:100px;"><label>Qtd.</label><input type="text" inputmode="decimal" id="calcQtd" value="'+escapeHtml(calc.qtd)+'"></div>'
          + '<div class="field narrow" style="max-width:150px;"><label>Margem que quero (%)</label><input type="text" inputmode="decimal" id="calcMargem" value="'+escapeHtml(calc.margem)+'"></div>'
        + '</div>'
        + '<div class="inline-form">'
          + '<div class="field narrow"><label>Imposto (%) <span class="opt">da empresa</span></label><input type="text" inputmode="decimal" id="calcImposto" value="'+escapeHtml(impostoVal)+'"></div>'
          + '<div class="field narrow"><label>Taxa do marketplace (%) <span class="opt">opcional</span></label><input type="text" inputmode="decimal" id="calcTaxaPct" value="'+escapeHtml(calc.taxaPct)+'"></div>'
          + '<div class="field narrow"><label>Taxa fixa (R$) <span class="opt">opcional</span></label><input type="text" inputmode="decimal" id="calcTaxaFixa" value="'+escapeHtml(calc.taxaFixa)+'"></div>'
          + '<div class="field narrow"><label>Frete que eu pago (R$) <span class="opt">opcional</span></label><input type="text" inputmode="decimal" id="calcFrete" value="'+escapeHtml(calc.frete)+'"></div>'
        + '</div>'
        + '<div id="calcResultado" style="margin-top:6px;">'+calcResultadoHTML()+'</div>'
      + '</div>';

    function bindCalcPreco(){
      const ligar = (id, chave)=>{
        const el = document.getElementById(id);
        if(!el) return;
        el.addEventListener('input', ()=>{
          calc[chave] = el.value;
          const out = document.getElementById('calcResultado');
          if(out) out.innerHTML = calcResultadoHTML();
        });
      };
      ligar('calcBusca','busca'); ligar('calcQtd','qtd'); ligar('calcMargem','margem');
      ligar('calcTaxaPct','taxaPct'); ligar('calcTaxaFixa','taxaFixa'); ligar('calcFrete','frete');
      ligar('calcImposto','impostoEditado');
    }

    async function loadCalcDados(){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId){ calc.produtos = []; return; }
      if(calc.empresaId !== filtro.empresaId){ calc.empresaId = filtro.empresaId; calc.impostoEditado = null; calc.busca = ''; }
      try{
        calc.produtos = await apiProdutosAtivos(filtro.empresaId);
        const res = await fetch('/api/config-financeiro/'+filtro.empresaId);
        const body = await res.json().catch(()=>({}));
        calc.impostoPct = res.ok && body && Number.isFinite(Number(body.aliquotaImposto)) ? Number(body.aliquotaImposto) : 0;
      }catch(e){ /* sem produtos a calculadora só mostra a dica — nunca quebra a tela */ }
    }

    function load(){ loadCalcDados().then(render); loadVendas(); }

    window.CerneFiltro.onChange(()=>{
      const page = document.getElementById('page-vendaBalcao');
      if(page && page.classList.contains('active')){ loadCalcDados().then(render); loadVendas(); }
      else render();
    });

    // ---- cards de resumo do período (soma simples do que já veio na
    // listagem — nunca um cálculo novo/paralelo ao de lib/relatorioVendas.js,
    // só uma soma de exibição das próprias vendas de balcão) ----
    function cardsHTML(){

      const totalVendido = validas.reduce((acc,v)=> round2(acc+v.valorTotal), 0);
      const totalCusto = validas.reduce((acc,v)=> v.custoProdutoTotal===null ? acc : round2(acc+v.custoProdutoTotal), 0);
      const temCustoIncompleto = validas.some(v=>v.custoProdutoTotal===null);
      const margem = temCustoIncompleto ? null : round2(totalVendido - validas.reduce((acc,v)=>round2(acc+v.desconto),0) - totalCusto);
      const items = [
        { label:'Vendido no período', icon:'wallet', tint:'success', valueHtml: money(totalVendido), note: validas.length+' venda(s) válida(s)' },
        { label:'Margem estimada', icon:'checkCircle', tint: margem===null ? 'muted' : (margem>=0?'success':'danger'),
          valueHtml: margem===null ? '—' : money(margem), note: temCustoIncompleto ? 'Parcial — há item sem custo cadastrado' : 'Vendido − desconto − custo do produto' },
        { label:'Cancelada(s)', icon:'x', tint:'muted', valueHtml: String(state.vendas.length - validas.length), note:'Nunca apagada — fica registrada como cancelada' },
      ];
      return '<div class="kpi-hero-grid">' + items.map(k=>
        '<div class="kpi-hero" style="'+(k.tint==='muted'?'--tint:var(--text-muted);--tint-soft:var(--surface-2);':'--tint:var(--'+k.tint+');--tint-soft:var(--'+k.tint+'-soft);')+'">'
        + '<div class="kpi-hero-top"><div class="kpi-hero-icon">'+svg(k.icon)+'</div></div>'
        + '<div class="kpi-hero-label">'+escapeHtml(k.label)+'</div>'
        + '<div class="kpi-hero-value">'+k.valueHtml+'</div>'
        + '<div class="kpi-hero-foot"><span class="kpi-hero-note">'+escapeHtml(k.note)+'</span></div>'
      + '</div>').join('') + '</div>';
    }

      const root = document.getElementById('vendaBalcaoRoot');
      if(!root) return;
      const filtro = window.CerneFiltro.state;

      if(!filtro.loadingEmpresas && !filtro.empresas.length){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
          + '<h2>Nenhuma empresa ativa</h2>'
          + '<p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> antes de registrar uma venda de balcão.</p></div>';
        return;
      }

      let html = '<div class="field-hint" style="margin-bottom:12px;max-width:820px;">Venda feita presencialmente, na sua empresa, pro cliente final — sempre à vista. Depois de registrada, ela conta como faturamento de verdade: aparece na Visão Geral, no DRE, nos Relatórios e no Faturamento, e o estoque "fora do Full" já sai descontado.</div>';

      html += calcPrecoHTML();

      html += '<div class="toolbar"><span></span><div class="search-box">'+svg('search')+'<input type="text" id="vbSearch" placeholder="Buscar por cliente ou número da venda" value="'+state.search.replace(/"/g,'&quot;')+'"></div></div>';

      if(state.loading || filtro.loadingEmpresas){

        html += '<div class="empty-panel"><div class="empty-icon">'+svg('store')+'</div>'

      } else {
        html += cardsHTML();
        html += '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'

          + state.vendas.map(v => '<tr'+(v.cancelado?' style="opacity:.6;"':'')+'>'
            + '<td class="tab-nums">'+fmtDateTime(v.dataVenda)+'</td>'
            + '<td class="mono">VB-'+String(v.numeroVenda).padStart(6,'0')+'</td>'
            + '<td>'+escapeHtml(v.clienteNome||'—')+'</td>'
            + '<td class="cell-stack-sub">'+v.itens.map(it=>escapeHtml(it.titulo)+' ×'+it.quantidade).join(', ')+'</td>'
            + '<td>'+(FORMA_PAGAMENTO_LABEL[v.formaPagamento]||escapeHtml(v.formaPagamento))+'</td>'

            + '<td class="mono">'+money(v.valorTotal)+'</td>'
            + '<td>'+(v.cancelado ? '<span class="badge-status off"><span class="dot"></span>Cancelada</span>' : '<span class="badge-status on"><span class="dot"></span>Concluída</span>')+'</td>'
            + '<td><div class="row-actions">'+(v.cancelado?'':'<button class="row-ic-btn danger" data-cancelar="'+v.id+'" title="Cancelar venda" type="button">'+svg('x')+'</button>')+'</div></td>'

          + '</tbody></table></div></div>';
      }

      root.innerHTML = html;
      bindCalcPreco();

      const search = document.getElementById('vbSearch');
      if(search) search.addEventListener('input', ()=>{
        state.search = search.value;
        clearTimeout(search._t);
        search._t = setTimeout(loadVendas, 300);
      });
      root.querySelectorAll('[data-cancelar]').forEach(b=> b.addEventListener('click', ()=> cancelarVenda(Number(b.dataset.cancelar))));
    }

    async function cancelarVenda(id){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId) return;
      if(!window.confirm('Cancelar esta venda de balcão? Ela nunca é apagada — fica registrada como cancelada e sai do faturamento.')) return;
      try{
        await api('/'+id+'/cancelar', { method:'PATCH', body: JSON.stringify({ empresaId: filtro.empresaId }) });
        toast('Venda cancelada.');
        loadVendas();
      }catch(err){ toast(err.message, 'error'); }
    }

    // ---- modal "Nova venda": carrinho de itens do catálogo já cadastrado
    // (nunca produto digitado solto), com total/margem calculados ao vivo
    // no navegador — só pra conferência visual; o valor que É gravado
    // sempre vem recalculado no servidor (lib/vendaBalcao.js), nunca do
    // que o navegador mandar pronto. ----
    let carrinho = [];
    let produtosCatalogo = [];

    function carrinhoTotais(){
      const bruto = carrinho.reduce((acc,it)=> round2(acc + round2(it.quantidade*it.precoUnitarioVenda)), 0);
      const custoTotal = carrinho.reduce((acc,it)=> it.custo===null ? acc : round2(acc + round2(it.quantidade*it.custo)), 0);
      const temSemCusto = carrinho.some(it=>it.custo===null);
      return { bruto, custoTotal, temSemCusto };
    }

    // 05/10/2026, pedido explícito do usuário: na tabela da venda, ao digitar
    // o PREÇO de venda aparece a MARGEM %; ao digitar a MARGEM % o PREÇO é
    // calculado. Margem % = (preço − custo) ÷ preço — a mesma margem da
    // linha "Margem estimada" abaixo (venda − custo; balcão não tem taxa de
    // marketplace). Preço pela margem: custo ÷ (1 − margem%). Os campos são
    // atualizados NO LUGAR (sem refazer a tabela) pra o cursor não sair do
    // campo a cada tecla digitada.
    function fmtNumInput(n){ return String(n).replace('.',','); }
    function margemPctDe(it){
      if(it.custo===null || it.custo===undefined || !(it.precoUnitarioVenda>0)) return null;
      return round2(((it.precoUnitarioVenda - it.custo) / it.precoUnitarioVenda) * 100);
    }
    function precoPelaMargem(custo, margemPct){
      if(custo===null || custo===undefined || !(margemPct<100)) return null;
      return Math.ceil((custo / (1 - margemPct/100)) * 100) / 100; // arredonda pra cima: nunca abaixo da margem pedida
    }

    function modalRenderTotais(){
      const totais = document.getElementById('vbTotais');
      if(!totais) return;
      const t = carrinhoTotais();
      const descontoInput = document.getElementById('vbDesconto');
      const desconto = descontoInput ? (Number((descontoInput.value||'0').replace(',','.')) || 0) : 0;
      const liquido = round2(t.bruto - desconto);
      const margem = t.temSemCusto ? null : round2(liquido - t.custoTotal);
      const margemPct = (margem===null || !(liquido>0)) ? null : round2((margem/liquido)*100);
      totais.innerHTML = '<div class="bd-row"><span class="bd-label">Subtotal dos itens</span><span class="bd-value">'+money(t.bruto)+'</span></div>'
        + '<div class="bd-row neg"><span class="bd-label">(−) Desconto</span><span class="bd-value">− '+money(desconto)+'</span></div>'
        + '<div class="bd-row total"><span class="bd-label">= Total da venda</span><span class="bd-value">'+money(liquido)+'</span></div>'
        + '<div class="bd-row total"><span class="bd-label">Margem estimada</span><span class="bd-value '+(margem===null?'':(margem>=0?'positive':'negative'))+'">'+(margem===null?'Parcial — item sem custo cadastrado':money(margem)+(margemPct!==null?' ('+pctFmtBR(margemPct)+')':''))+'</span></div>';
    }
    function pctFmtBR(v){ return Number(v).toLocaleString('pt-BR',{minimumFractionDigits:1, maximumFractionDigits:2})+'%'; }

    // Atualiza só as células calculadas de UMA linha (total e, se pedido, o
    // campo de margem/preço), sem recriar os inputs.
    function modalAtualizarLinha(idx, quem){
      const it = carrinho[idx];
      const tr = document.querySelector('#vbCarrinhoBody tr[data-row="'+idx+'"]');
      if(!tr || !it) return;
      const totalEl = tr.querySelector('.vb-total');
      if(totalEl) totalEl.textContent = money(round2(it.quantidade*it.precoUnitarioVenda));
      if(quem!=='margem'){
        const m = margemPctDe(it);
        const mEl = tr.querySelector('.vb-margem');
        if(mEl) mEl.value = m===null ? '' : fmtNumInput(m);
      }
      if(quem!=='preco'){
        const pEl = tr.querySelector('.vb-preco');
        if(pEl) pEl.value = it.precoUnitarioVenda ? fmtNumInput(it.precoUnitarioVenda) : '';
      }
    }

    function modalRenderCarrinho(){
      const tbody = document.getElementById('vbCarrinhoBody');
      if(!tbody) return;
      if(!carrinho.length){
        tbody.innerHTML = '<tr><td colspan="7" style="text-align:center;color:var(--text-faint);padding:18px 0;">Nenhum produto adicionado ainda.</td></tr>';
      } else {
        tbody.innerHTML = carrinho.map((it,idx)=> {
          const m = margemPctDe(it);
          const semCusto = it.custo===null || it.custo===undefined;
          return '<tr data-row="'+idx+'">'
          + '<td>'+escapeHtml(it.nome)+'<div class="cell-stack-sub mono">SKU '+escapeHtml(it.sku)+'</div></td>'
          + '<td style="width:80px;"><input type="number" min="0.001" step="0.001" class="vb-qtd" data-idx="'+idx+'" value="'+it.quantidade+'" style="width:70px;"></td>'
          + '<td class="mono" style="width:90px;">'+(semCusto?'<span class="dash">sem custo</span>':money(it.custo))+'</td>'
          + '<td style="width:120px;"><input type="text" inputmode="decimal" class="vb-preco" data-idx="'+idx+'" value="'+(it.precoUnitarioVenda?fmtNumInput(it.precoUnitarioVenda):'')+'" placeholder="0,00" style="width:100px;"></td>'
          + '<td style="width:100px;"><input type="text" inputmode="decimal" class="vb-margem" data-idx="'+idx+'" value="'+(m===null?'':fmtNumInput(m))+'" placeholder="%"'+(semCusto?' disabled title="Cadastre o custo do produto pra usar a margem"':'')+' style="width:80px;"></td>'
          + '<td class="mono vb-total">'+money(round2(it.quantidade*it.precoUnitarioVenda))+'</td>'
          + '<td><button type="button" class="row-ic-btn danger" data-remover-item="'+idx+'">'+svg('trash')+'</button></td>'
        + '</tr>'; }).join('');
      }
      modalRenderTotais();
    }

    function abrirModalNovaVenda(){
      const filtro = window.CerneFiltro.state;
      if(!filtro.empresaId){ toast('Selecione uma empresa no filtro do topo antes de registrar uma venda.', 'error'); return; }
      carrinho = [];
      const modalRoot = document.getElementById('modalRoot');
      modalRoot.innerHTML = '<div class="modal-overlay" id="modalOverlay"><div class="modal-card wide"><div class="table-skel"><span class="spin"></span><span>Carregando produtos…</span></div></div></div>';
      const overlayEarly = document.getElementById('modalOverlay');
      overlayEarly.addEventListener('click', (e)=>{ if(e.target===overlayEarly) modalRoot.innerHTML=''; });

      apiProdutosAtivos(filtro.empresaId).then((produtos)=>{
        produtosCatalogo = produtos;
        if(!produtos.length){
          modalRoot.innerHTML = '<div class="modal-overlay" id="modalOverlay"><div class="modal-card"><div class="empty-panel"><div class="empty-icon">'+svg('box')+'</div><h2>Nenhum produto ativo cadastrado</h2><p class="lead">Cadastre pelo menos um produto em <strong>Cadastros → Produtos</strong> antes de registrar uma venda de balcão.</p></div><div class="modal-actions"><button type="button" class="btn-secondary" id="modalCancel">Fechar</button></div></div></div>';
          document.getElementById('modalCancel')?.addEventListener('click', ()=> modalRoot.innerHTML='');
          document.getElementById('modalOverlay')?.addEventListener('click', (e)=>{ if(e.target.id==='modalOverlay') modalRoot.innerHTML=''; });
          return;
        }
        renderModalNovaVenda();
      }).catch((err)=>{ toast(err.message, 'error'); modalRoot.innerHTML=''; });
    }

    function renderModalNovaVenda(){
      const modalRoot = document.getElementById('modalRoot');

            + '<div class="field"><label>Caixa / produto <span class="opt">(digite o SKU ou o nome)</span></label>'

            + '<th>Produto</th><th>Qtd.</th><th>Custo unit.</th><th>Preço unit. (R$)</th><th>Margem (%)</th><th>Total</th><th></th>'

          + '<div class="inline-form">'

            + '<div class="field narrow"><label>Cliente <span class="opt">(opcional)</span></label><input type="text" id="vbCliente" maxlength="200"></div>'
          + '</div>'
          + '<div class="field"><label>Observação <span class="opt">(opcional)</span></label><input type="text" id="vbObservacao"></div>'

          + '<div id="vbTotais" style="margin:10px 0;"></div>'
          + '<div class="field-error" id="vbErro" style="display:none;"></div>'
          + '<div class="modal-actions">'

      // Acha o produto pelo que foi digitado: "SKU — nome" (item da lista),
      // SKU exato, nome exato ou — se só UM produto contém o texto — esse.
      function achaProdutoDigitado(txt){
        const t = String(txt||'').trim().toLowerCase();
        if(!t) return null;
        let p = produtosCatalogo.find(x=> (x.sku+' — '+x.nome).toLowerCase()===t) || produtosCatalogo.find(x=> String(x.sku).trim().toLowerCase()===t);
        if(p) return p;
        const nomes = produtosCatalogo.filter(x=> String(x.nome).trim().toLowerCase()===t);
        if(nomes.length===1) return nomes[0];
        const parecidos = produtosCatalogo.filter(x=> (x.sku+' '+x.nome).toLowerCase().includes(t));
        return parecidos.length===1 ? parecidos[0] : null;
      }
      function adicionarProdutoDigitado(silencioso){
        const buscaEl = document.getElementById('vbProdutoBusca');
        const qtdEl = document.getElementById('vbProdutoQtd');
        const produto = achaProdutoDigitado(buscaEl.value);
        if(!produto){ if(!silencioso) toast(buscaEl.value.trim() ? 'Não achei essa caixa/produto. Escolha uma da lista que aparece ao digitar.' : 'Digite o SKU ou o nome da caixa.', 'error'); return; }
        const quantidade = Number(String(qtdEl.value).replace(',','.'));
        if(!Number.isFinite(quantidade) || quantidade<=0){ toast('Informe uma quantidade válida.', 'error'); return; }
        const existente = carrinho.find(it=>it.sku===produto.sku);
        let idx;
        if(existente){ existente.quantidade = round2(existente.quantidade + quantidade); idx = carrinho.indexOf(existente); }
        else { carrinho.push({ sku:produto.sku, nome:produto.nome, custo:produto.custo, quantidade, precoUnitarioVenda:0 }); idx = carrinho.length-1; }
        buscaEl.value=''; qtdEl.value='1';
        modalRenderCarrinho();
        const novoPreco = document.querySelector('#vbCarrinhoBody tr[data-row="'+idx+'"] .vb-preco');
        if(novoPreco) novoPreco.focus();
      }
      document.getElementById('vbAdicionarItem')?.addEventListener('click', ()=> adicionarProdutoDigitado(false));
      const buscaEl0 = document.getElementById('vbProdutoBusca');
      // Enter no campo de busca adiciona o produto (e NÃO finaliza a venda).
      buscaEl0.addEventListener('keydown', (e)=>{ if(e.key==='Enter'){ e.preventDefault(); adicionarProdutoDigitado(false); } });
      // Escolheu um item da lista (texto idêntico a "SKU — nome") → já adiciona.
      buscaEl0.addEventListener('change', ()=>{
        const p = achaProdutoDigitado(buscaEl0.value);
        if(p && (p.sku+' — '+p.nome).toLowerCase()===buscaEl0.value.trim().toLowerCase()) adicionarProdutoDigitado(true);
      });
      buscaEl0.focus();

      document.getElementById('vbCarrinhoBody')?.addEventListener('input', (e)=>{
        const idx = Number(e.target.dataset.idx);
        if(Number.isNaN(idx) || !carrinho[idx]) return;
        const it = carrinho[idx];
        const num = (s)=>{ const t=String(s).trim(); if(!t) return NaN; const n = Number(t.indexOf(',')>=0 ? t.replace(/\./g,'').replace(',','.') : t); return n; };
        if(e.target.classList.contains('vb-qtd')){
          const v = Number(e.target.value);
          if(Number.isFinite(v) && v>0) it.quantidade = v;
          modalAtualizarLinha(idx, 'qtd');
        } else if(e.target.classList.contains('vb-preco')){
          const v = num(e.target.value);
          it.precoUnitarioVenda = Number.isFinite(v) && v>0 ? v : 0;
          modalAtualizarLinha(idx, 'preco');   // recalcula a margem %
        } else if(e.target.classList.contains('vb-margem')){
          const v = num(e.target.value);
          const preco = Number.isFinite(v) ? precoPelaMargem(it.custo, v) : null;
          if(preco!==null){ it.precoUnitarioVenda = preco; }
          modalAtualizarLinha(idx, 'margem');  // recalcula o preço
        }
        modalRenderTotais();
      });
      document.getElementById('vbCarrinhoBody')?.addEventListener('click', (e)=>{
        const btn = e.target.closest('[data-remover-item]');
        if(!btn) return;
        carrinho.splice(Number(btn.dataset.removerItem), 1);
        modalRenderCarrinho();
      });
      document.getElementById('vbDesconto')?.addEventListener('input', modalRenderTotais);

      document.getElementById('vbForm')?.addEventListener('submit', async (e)=>{
        e.preventDefault();
        const erroEl = document.getElementById('vbErro');
        erroEl.style.display='none'; erroEl.textContent='';

        if(!carrinho.length){ erroEl.textContent='Adicione ao menos um produto à venda.'; erroEl.style.display='block'; return; }
        if(carrinho.some(it=>!(it.precoUnitarioVenda>0))){ erroEl.textContent='Informe o preço de venda de cada produto (maior que zero).'; erroEl.style.display='block'; return; }

        const descontoRaw = document.getElementById('vbDesconto').value;
        const payload = {
          empresaId: window.CerneFiltro.state.empresaId,
          formaPagamento: document.getElementById('vbFormaPagamento').value,
          clienteNome: document.getElementById('vbCliente').value,
          observacao: document.getElementById('vbObservacao').value,
          desconto: descontoRaw ? Number(String(descontoRaw).replace(',','.')) : 0,
          itens: carrinho.map(it=>({ sku:it.sku, quantidade:it.quantidade, precoUnitarioVenda:it.precoUnitarioVenda })),
        };

        const saveBtn = document.getElementById('modalSave');
        saveBtn.disabled = true;
        try{
          await api('', { method:'POST', body: JSON.stringify(payload) });
          close();
          toast('Venda registrada — já entrou no faturamento e o estoque foi baixado.');
          loadVendas();
        }catch(err){
          saveBtn.disabled = false;
          erroEl.textContent = err.message || 'Não foi possível registrar a venda.';
          erroEl.style.display='block';
        }
      });
    }

    const btnNovaVenda = document.getElementById('btnNovaVendaBalcao');
    if(btnNovaVenda) btnNovaVenda.addEventListener('click', abrirModalNovaVenda);

    window.VendaBalcao = { load };
  })();
