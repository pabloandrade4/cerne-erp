  /* ================= AGENTES DE IA — Fase 1 (14/09/2026) =================
     Pedido explícito do usuário: uma aba dedicada "Agentes IA", com cada
     agente separado — preparado para, no futuro, o usuário conseguir
     "conversar com cada um separado" (nenhuma conversa é implementada
     agora, só a identidade de cada agente, vinda de GET /api/ia-agentes).
     Cada agente mostra: identidade, as sugestões PENDENTES (aprovar/
     alterar/recusar) e o HISTÓRICO de decisões já tomadas + resultado real
     quando já avaliado. Fluxo: DADOS → ANÁLISE → RECOMENDAÇÃO (já rodando
     em segundo plano, ver lib/ia/adsDecisoesCiclo.js e
     lib/ia/promocoesDecisoesStore.js) → DECISÃO DO USUÁRIO (aqui) →
     RESULTADO → aprendizado (Fase 2, ainda não implementada).
     IMPORTANTE: nenhuma decisão é executada no Mercado Livre nesta fase —
     aprovar só REGISTRA a decisão (ver rotas .../decisoes no backend).
     Os dois agentes (Ads e Performance, Promoções) usam a mesma estrutura
     de tela — criarModuloAgenteDecisoes() monta cada um com seus próprios
     endpoints/rótulos, pra nunca duplicar a lógica entre os dois. */
  (function(){
    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function money(v){ return v===null||v===undefined ? '—' : 'R$ ' + Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2}); }
    function pctFmt(v){ return v===null||v===undefined ? '—' : Number(v).toLocaleString('pt-BR',{minimumFractionDigits:1, maximumFractionDigits:1})+'%'; }
    function fmtDataHora(v){
      if(!v) return '—';
      try{ const d = new Date(v); return d.toLocaleDateString('pt-BR') + ' ' + d.toLocaleTimeString('pt-BR', { hour:'2-digit', minute:'2-digit' }); }
      catch(e){ return '—'; }
    }
    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+escapeHtml(message)+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 4200);
    }

    const STATUS_LABEL = { pendente:'Pendente', aprovada:'Aprovada', alterada:'Alterada', recusada:'Recusada', expirada:'Expirada' };
    const STATUS_COR = { pendente:'var(--warning)', aprovada:'var(--success)', alterada:'var(--copper)', recusada:'var(--danger)', expirada:'var(--text-faint)' };

    const TIPO_ACAO_LABEL = {
      pausar_anuncio: 'Pausar anúncio',
      colocar_sku_em_campanha: 'Considerar incluir em campanha',
      pausar_campanha: 'Pausar campanha',
      diminuir_orcamento: 'Diminuir orçamento',
      aumentar_orcamento: 'Aumentar orçamento',
      ativar_campanha: 'Ativar campanha',
      entrar_promocao: 'Entrar na promoção',
      nao_entrar_promocao: 'Não entrar na promoção',
      sair_promocao: 'Sair da promoção',
      revisar_preco: 'Revisar preço promocional',
      estoque_alto: 'Estoque alto — considerar promoção',
    };

    function sugestaoResumoHTML(d){
      const v = d.valorSugeridoIa || {};
      const label = TIPO_ACAO_LABEL[d.tipoAcao] || d.tipoAcao;
      let detalhe = '';
      if(v.orcamentoAtual !== undefined && v.orcamentoSugerido !== undefined){
        detalhe = ' — de ' + money(v.orcamentoAtual) + ' para ' + money(v.orcamentoSugerido);
      } else if(v.precoPromoAtual !== undefined && v.precoPromoAtual !== null){
        detalhe = ' — preço promocional atual: ' + money(v.precoPromoAtual);
      }
      return '<div><strong>'+escapeHtml(label)+'</strong>'+escapeHtml(detalhe)+'</div>'
        + '<div class="cell-stack-sub" style="margin-top:3px;color:var(--text-muted);">'+escapeHtml(d.motivo||'')+'</div>';
    }

    function decisaoResumoHTML(d){
      const cor = STATUS_COR[d.statusDecisao] || 'var(--text-faint)';
      const label = STATUS_LABEL[d.statusDecisao] || d.statusDecisao;
      let valorTxt = '';
      const vd = d.valorDecididoUsuario;
      if(vd && vd.orcamentoDecidido !== undefined) valorTxt = ' — ' + money(vd.orcamentoDecidido);
      else if(vd && vd.precoPromoDecidido !== undefined) valorTxt = ' — ' + money(vd.precoPromoDecidido);
      return '<span style="font-weight:700;color:'+cor+';">'+escapeHtml(label)+'</span>'+escapeHtml(valorTxt)
        + (d.decididoEm ? '<div class="cell-stack-sub" style="margin-top:2px;">'+fmtDataHora(d.decididoEm)+(d.decididoPor?(' · '+escapeHtml(d.decididoPor)):'')+'</div>' : '');
    }

// Ações de campanha que já têm execução real ligada ao Mercado Livre
    // (19/09/2026, ver lib/ia/adsExecutor.js) — espelha
    // ACOES_EXECUTAVEIS de lá; as demais (pausar_anuncio,
    // colocar_sku_em_campanha) continuam só recomendação nesta fase.
    const ACOES_EXECUTAVEIS_ML = ['pausar_campanha','ativar_campanha','diminuir_orcamento','aumentar_orcamento'];

    function executadoHTML(d, cfg){
      if(d.statusDecisao === 'recusada' || d.statusDecisao === 'expirada') return '<span style="color:var(--text-faint);">—</span>';
      if(!cfg || !cfg.executaDeVerdade){
        return '<span style="color:var(--text-faint);font-style:italic;" title="Fase 1: a IA só registra a decisão, nunca executa automaticamente no Mercado Livre.">Não executado (Fase 1)</span>';
      }
      if(!ACOES_EXECUTAVEIS_ML.includes(d.tipoAcao)){
        return '<span style="color:var(--text-faint);font-style:italic;" title="Este tipo de sugestão ainda não tem execução automática — continua só recomendação.">Só recomendação</span>';
      }
      if(d.executado){
        return '<span style="color:var(--success);font-weight:600;" title="'+escapeHtml('Aplicado de verdade no Mercado Livre em '+fmtDataHora(d.executadoEm))+'">✅ Executado</span>';
      }
      if(d.execucaoErro){
        return '<span style="color:var(--danger);cursor:help;" title="'+escapeHtml(d.execucaoErro)+'">⚠️ Não executado</span>';
      }
      return '<span style="color:var(--text-faint);font-style:italic;">Aguardando execução</span>';
    }

    function podeAlterarNumerico(d){
      const v = d.valorSugeridoIa || {};
      return v.orcamentoSugerido !== undefined || d.tipoAcao === 'revisar_preco';
    }
    function campoAlteracaoChave(d){ return d.tipoAcao === 'revisar_preco' ? 'precoPromoDecidido' : 'orcamentoDecidido'; }
    function campoAlteracaoLabel(d){ return d.tipoAcao === 'revisar_preco' ? 'Novo preço promocional decidido (R$)' : 'Orçamento decidido (R$)'; }
    function valorInicialAlteracao(d){
      const v = d.valorSugeridoIa || {};
      return v.orcamentoSugerido !== undefined ? v.orcamentoSugerido : '';
    }

    function criarModuloAgenteDecisoes(cfg){
      const state = { agente:null, aba:'pendentes', itens:[], loading:false, error:null, editandoId:null, gerando:false, integStatus:null, integSalvando:false };

      async function apiAgente(){
        const res = await fetch('/api/ia-agentes');
        const body = await res.json().catch(()=>null);
        if(!res.ok) return null;
        return ((body && body.agentes) || []).find((a)=> a.codigo === cfg.codigoAgente) || null;
      }
      // Card "Integração Mercado Livre — execução automática" (19/09/2026,
      // pedido explícito do usuário: "eu vou aprovar, aí vai fazer... tem
      // que mudar as configurações do developers" — ver
      // lib/ia/adsExecutor.js/routes/ads.js). Só existe pra agentes que já
      // executam de verdade (cfg.execucao definido) — Promoções continua
      // sem esta seção, sem nenhuma mudança de comportamento pra ela.
      async function apiStatusIntegracao(empresaId){
        const res = await fetch(cfg.execucao.endpointStatusIntegracao + '?empresaId=' + empresaId);
        const body = await res.json().catch(()=>null);
        if(!res.ok) throw new Error((body && body.error) || 'Não foi possível carregar o status da integração.');
        return body;
      }
      async function apiSalvarPermiteEscrita(empresaId, permiteEscritaMl){
        const configAtual = await fetch(cfg.execucao.endpointConfig + '?empresaId=' + empresaId).then(r=>r.json()).catch(()=>({}));
        const res = await fetch(cfg.execucao.endpointConfig, {
          method:'PUT', headers:{'Content-Type':'application/json'},
          body: JSON.stringify({ empresaId, margemMinimaPct: configAtual.margemMinimaPct, permiteEscritaMl }),
        });
        const body = await res.json().catch(()=>null);
        if(!res.ok) throw new Error((body && body.error) || 'Não foi possível salvar.');
        return body;
      }
      async function carregarIntegracao(){
        if(!cfg.execucao) return;
        const filtro = window.CerneFiltro.state;
        if(!filtro.empresaId) return;
        try{ state.integStatus = await apiStatusIntegracao(filtro.empresaId); }
        catch(e){ state.integStatus = null; }
        render();
      }
      async function alternarPermiteEscrita(ligar){
        const filtro = window.CerneFiltro.state;
        if(!filtro.empresaId) return;
        state.integSalvando = true; render();
        try{
          await apiSalvarPermiteEscrita(filtro.empresaId, ligar);
          toast(ligar ? 'Execução automática de Ads ligada.' : 'Execução automática de Ads desligada.');
          await carregarIntegracao();
        }catch(e){ toast(e.message || 'Falha ao salvar.', 'error'); }
        finally{ state.integSalvando = false; render(); }
      }
      function cardIntegracaoExecucaoHTML(){
        if(!cfg.execucao) return '';
        const r = state.integStatus && state.integStatus.resumo;
        const ligado = !!(r && r.escritaDisponivel);
        return '<div class="table-card" style="padding:20px;margin-bottom:14px;">'
          + '<div class="panel-title" style="margin-bottom:10px;"><span class="panel-title-ic">'+svg('link')+'</span>Integração Mercado Livre — execução automática</div>'
          + '<div class="field-hint" style="margin-bottom:12px;">'+escapeHtml((r && r.mensagem) || 'Carregando status da integração…')+'</div>'
          + '<label style="display:flex;align-items:center;gap:8px;'+(state.integSalvando?'opacity:.6;':'cursor:pointer;')+'">'
            + '<input type="checkbox" id="'+cfg.rootElId+'ToggleEscrita" '+(ligado?'checked':'')+' '+(state.integSalvando?'disabled':'')+'>'
            + '<span>Permitir que a IA execute de verdade no Mercado Livre quando eu aprovar uma sugestão</span>'
          + '</label>'
          + '<div class="field-hint" style="margin-top:8px;color:var(--text-faint);">Antes de ligar: confirme no painel do Mercado Livre Developers que o aplicativo tem permissão de <strong>escrita</strong> (não só leitura) e reconecte a conta em Marketplaces — um token já emitido não ganha permissão nova sozinho.</div>'
        + '</div>';
      }
      async function apiDecisoes(empresaId, status){
        const params = new URLSearchParams({ empresaId, status: status || 'todas' });
        const res = await fetch(cfg.endpointDecisoes + '?' + params.toString());
        const body = await res.json().catch(()=>null);
        if(!res.ok) throw new Error((body && body.error) || 'Não foi possível carregar as decisões.');
        return (body && body.decisoes) || [];
      }
      async function apiDecidir(id, statusDecisao, valorDecididoUsuario){
        const res = await fetch(cfg.endpointDecidir(id), {
          method:'PUT', headers:{'Content-Type':'application/json'},
          body: JSON.stringify({ statusDecisao, valorDecididoUsuario, decididoPor: 'Pablo' }),
        });
        const body = await res.json().catch(()=>null);
        if(!res.ok) throw new Error((body && body.error) || 'Não foi possível registrar a decisão.');
        return body;
      }
      async function apiGerarAgora(empresaId){
        const res = await fetch(cfg.endpointGerarAgora, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ empresaId }) });
        const body = await res.json().catch(()=>null);
        if(!res.ok) throw new Error((body && body.error) || 'Falha ao gerar sugestões agora.');
        return body;
      }

      async function carregarDecisoes(){
        const filtro = window.CerneFiltro.state;
        if(!filtro.empresaId){ state.itens = []; state.loading = false; render(); return; }
        state.loading = true; state.error = null; render();
        try{ state.itens = await apiDecisoes(filtro.empresaId, state.aba === 'pendentes' ? 'pendente' : 'decidida'); }
        catch(e){ state.error = e.message; state.itens = []; }
        state.loading = false; render();
      }

      async function gerarAgora(){
        const filtro = window.CerneFiltro.state;
        if(!filtro.empresaId) return;
        state.gerando = true; render();
        try{
          await apiGerarAgora(filtro.empresaId);
          toast('Sugestões atualizadas.');
          await carregarDecisoes();
        }catch(e){ toast(e.message || 'Falha ao gerar sugestões agora.', 'error'); }
        finally{ state.gerando = false; render(); }
      }

      async function decidir(id, statusDecisao, valorDecididoUsuario){
        try{
          await apiDecidir(id, statusDecisao, valorDecididoUsuario);
          toast(statusDecisao === 'recusada' ? 'Sugestão recusada.' : 'Decisão registrada.');
          state.editandoId = null;
          await carregarDecisoes();
        }catch(e){ toast(e.message || 'Falha ao registrar decisão.', 'error'); }
      }

      async function load(){
        if(!state.agente) state.agente = await apiAgente();
        await Promise.all([carregarDecisoes(), carregarIntegracao()]);
      }

      window.CerneFiltro.onChange(()=>{
        const page = document.getElementById('page-'+cfg.pageId);
        if(page && page.classList.contains('active')){ carregarDecisoes(); carregarIntegracao(); }
        else render();
      });

      function tabelaPendentesHTML(itens){
        let html = '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>Atualizado em</th><th>'+escapeHtml(cfg.colunaReferenciaLabel)+'</th><th>Sugestão da IA (situação e motivo)</th><th></th>'
          + '</tr></thead><tbody>';
        itens.forEach((d)=>{
          let acoes = '<button class="row-ic-btn" data-acao="aprovada" data-id="'+d.id+'" type="button" title="Aprovar — aceitar exatamente o que a IA sugeriu">'+svg('checkCircle')+'</button>';
          if(podeAlterarNumerico(d)) acoes += ' <button class="row-ic-btn" data-acao="alterar-toggle" data-id="'+d.id+'" type="button" title="Alterar o valor sugerido">'+svg('edit')+'</button>';
          acoes += ' <button class="row-ic-btn" data-acao="recusada" data-id="'+d.id+'" type="button" title="Recusar">'+svg('x')+'</button>';

          html += '<tr>'
            + '<td class="mono" style="white-space:nowrap;">'+fmtDataHora(d.atualizadoEm)+'</td>'
            + '<td>'+cfg.referenciaHTML(d)+'</td>'
            + '<td style="max-width:420px;">'+sugestaoResumoHTML(d)+'</td>'
            + '<td style="white-space:nowrap;"><div class="row-actions">'+acoes+'</div></td>'
          + '</tr>';

          if(state.editandoId === d.id){
            html += '<tr><td colspan="4" style="background:var(--surface-2);padding:12px 16px;">'
              + '<div class="inline-form" style="margin:0;align-items:flex-end;">'
                + '<div class="field narrow"><label>'+campoAlteracaoLabel(d)+'</label><input type="number" step="0.01" min="0" id="valorAlterado-'+d.id+'" value="'+escapeHtml(String(valorInicialAlteracao(d)))+'"></div>'
                + '<button class="primary-btn enabled" data-acao="confirmar-alterar" data-id="'+d.id+'" type="button">'+svg('check')+'<span>Confirmar alteração</span></button>'
                + '<button class="ghost-btn" data-acao="cancelar-alterar" data-id="'+d.id+'" type="button" style="cursor:pointer;opacity:1;"><span>Cancelar</span></button>'
              + '</div>'
            + '</td></tr>';
          }
        });
        return html + '</tbody></table></div></div>';
      }

      function tabelaHistoricoHTML(itens){
        return '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>'+escapeHtml(cfg.colunaReferenciaLabel)+'</th><th>Sugestão da IA (situação e motivo)</th><th>Minha decisão</th><th>Ação executada</th><th>Resultado posterior</th>'
          + '</tr></thead><tbody>'
          + itens.map((d)=> '<tr>'
            + '<td>'+cfg.referenciaHTML(d)+'</td>'
            + '<td style="max-width:380px;">'+sugestaoResumoHTML(d)+'</td>'
            + '<td>'+decisaoResumoHTML(d)+'</td>'
            + '<td>'+executadoHTML(d, cfg)+'</td>'
            + '<td>'+(d.resultadoAvaliadoEm ? cfg.resultadoHTML(d) : '<span style="color:var(--text-faint);font-style:italic;" title="A IA reavalia automaticamente alguns dias após a decisão.">Ainda não avaliado</span>')+'</td>'
          + '</tr>').join('')
          + '</tbody></table></div></div>';
      }

      function render(){
        const root = document.getElementById(cfg.rootElId);
        if(!root) return;
        const filtro = window.CerneFiltro.state;

        if(!filtro.loadingEmpresas && !filtro.empresas.length){
          root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
            + '<h2>Nenhuma empresa ativa</h2><p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> para usar este agente.</p></div>';
          return;
        }

        const a = state.agente;
        let html = '<div class="table-card" style="padding:20px;margin-bottom:14px;display:flex;gap:16px;align-items:flex-start;flex-wrap:wrap;">'
          + '<div class="kpi-hero-icon" style="flex-shrink:0;">'+svg((a && a.icone) || cfg.iconeFallback)+'</div>'
          + '<div style="min-width:0;flex:1;">'
            + '<div class="panel-title" style="margin:0 0 4px;">'+escapeHtml((a && a.nome) || cfg.nomeFallback)+'</div>'
            + '<div class="field-hint" style="max-width:760px;">'+escapeHtml((a && a.descricao) || cfg.descricaoFallback)+'</div>'
            + '<div class="field-hint" style="margin-top:8px;color:var(--text-faint);">'+(cfg.execucao
                ? 'A IA analisa e recomenda; ela nunca age sozinha. Quando você aprova uma sugestão, e a execução automática estiver ligada abaixo, a mudança já é aplicada de verdade no Mercado Livre. Toda decisão sua fica registrada no histórico.'
                : 'Fase 1: a IA só analisa e recomenda — nenhuma ação é executada automaticamente no Mercado Livre. Toda decisão sua fica registrada no histórico, e é isso que o agente vai usar para aprender.')+'</div>'
          + '</div>'
        + '</div>';

        html += cardIntegracaoExecucaoHTML();

        html += '<div class="toolbar">'
          + '<div class="seg" id="'+cfg.rootElId+'Tabs">'
            + '<button type="button" data-aba="pendentes" class="'+(state.aba==='pendentes'?'active':'')+'">Pendentes</button>'
            + '<button type="button" data-aba="historico" class="'+(state.aba==='historico'?'active':'')+'">Histórico</button>'
          + '</div>'
          + '<button class="ghost-btn" id="'+cfg.rootElId+'GerarAgora" type="button" style="cursor:pointer;opacity:1;" '+(state.gerando?'disabled':'')+'>'+svg('refreshCw')+'<span>'+(state.gerando?'Gerando…':'Gerar sugestões agora')+'</span></button>'
        + '</div>';

        if(state.loading){
          html += '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando…</span></div></div>';
        } else if(state.error){
          html += '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
            + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p></div>';
        } else if(!state.itens.length){
          html += '<div class="empty-panel"><div class="empty-icon">'+svg('checkCircle')+'</div>'
            + '<h2>'+(state.aba==='pendentes' ? 'Nenhuma sugestão pendente' : 'Nenhuma decisão no histórico ainda')+'</h2>'
            + '<p class="lead">'+(state.aba==='pendentes'
                ? 'Quando a IA identificar uma ação recomendada, ela aparece aqui para você aprovar, alterar ou recusar.'
                : 'Decisões que você aprovar, alterar ou recusar aparecem aqui, junto do resultado real assim que a IA reavaliar.')+'</p></div>';
        } else {
          html += state.aba === 'pendentes' ? tabelaPendentesHTML(state.itens) : tabelaHistoricoHTML(state.itens);
        }

        root.innerHTML = html;

        const toggleEscritaEl = document.getElementById(cfg.rootElId+'ToggleEscrita');
        if(toggleEscritaEl) toggleEscritaEl.addEventListener('change', ()=> alternarPermiteEscrita(toggleEscritaEl.checked));

        const tabsEl = document.getElementById(cfg.rootElId+'Tabs');
        if(tabsEl) tabsEl.querySelectorAll('button').forEach((b)=> b.addEventListener('click', ()=>{ state.aba = b.dataset.aba; state.editandoId = null; carregarDecisoes(); }));
        const btnGerar = document.getElementById(cfg.rootElId+'GerarAgora');
        if(btnGerar) btnGerar.addEventListener('click', gerarAgora);

        root.querySelectorAll('[data-acao]').forEach((btn)=> btn.addEventListener('click', ()=>{
          const id = Number(btn.dataset.id);
          const acao = btn.dataset.acao;
          if(acao === 'alterar-toggle'){ state.editandoId = state.editandoId === id ? null : id; render(); return; }
          if(acao === 'cancelar-alterar'){ state.editandoId = null; render(); return; }
          if(acao === 'confirmar-alterar'){
            const input = document.getElementById('valorAlterado-'+id);
            const valor = Number(input.value);
            if(!Number.isFinite(valor) || valor < 0){ toast('Informe um valor válido.', 'error'); return; }
            const d = state.itens.find((x)=> x.id === id);
            if(!d) return;
            decidir(id, 'alterada', { [campoAlteracaoChave(d)]: valor });
            return;
          }
          if(acao === 'aprovada'){
            const d = state.itens.find((x)=> x.id === id);
            decidir(id, 'aprovada', d ? d.valorSugeridoIa : null);
            return;
          }
          if(acao === 'recusada'){ decidir(id, 'recusada', null); return; }
        }));
      }

      return { load };
    }

    // ---- Agente "Ads e Performance" — layout dedicado (21/09/2026) ----
    // Pedido explícito do usuário: um mockup HTML pronto
    // (pf_ads_com_agente_layout.html) com DUAS abas dentro da mesma tela —
    // "Visão Ads" (KPIs, resumo da IA, status das campanhas, tabela por
    // anúncio) e "Agente de Ads" (status do agente, melhoria gerada,
    // impacto, o que fazer/o que o agente está fazendo/o que já melhorou,
    // linha do tempo, regras e aprovações) — "quero que faça completamente

      function metricasMelhora(){
        const av = avaliadas();
        let somaAntes = 0, somaDepois = 0, n = 0;
        av.forEach((d)=>{
          const antes = d.snapshot && d.snapshot.margemDepoisDoAdsPct;
          const depois = d.resultadoSnapshot && d.resultadoSnapshot.margemDepoisDoAdsPct;
          if(antes !== null && antes !== undefined && depois !== null && depois !== undefined){ somaAntes += Number(antes); somaDepois += Number(depois); n++; }

        if(!n) return null;
        return { antesPct: somaAntes/n, depoisPct: somaDepois/n, n };
      }

      function calcularImpacto(){
        const av = avaliadas();
        let lucroRecuperado = 0, economiaDesperdicio = 0, receitaEscala = 0, sucesso = 0, comparaveis = 0;
        av.forEach((d)=>{
          const s = d.snapshot || {}, r = d.resultadoSnapshot || {};
          if(s.margemDepoisDoAds !== null && s.margemDepoisDoAds !== undefined && r.margemDepoisDoAds !== null && r.margemDepoisDoAds !== undefined){
            const delta = Number(r.margemDepoisDoAds) - Number(s.margemDepoisDoAds);
            if(delta > 0) lucroRecuperado += delta;
            if(delta >= 0) sucesso++;
            comparaveis++;
          }
          if(BUCKET_ECONOMIA_ADS2.includes(d.tipoAcao) && s.investimento !== null && s.investimento !== undefined && r.investimento !== null && r.investimento !== undefined){
            const reducao = Number(s.investimento) - Number(r.investimento);
            if(reducao > 0) economiaDesperdicio += reducao;
          }
          if(BUCKET_ESCALA_ADS2.includes(d.tipoAcao) && s.faturamentoReal !== null && s.faturamentoReal !== undefined && r.faturamentoReal !== null && r.faturamentoReal !== undefined){
            const ganho = Number(r.faturamentoReal) - Number(s.faturamentoReal);
            if(ganho > 0) receitaEscala += ganho;
          }
        });
        return { lucroRecuperado, economiaDesperdicio, receitaEscala, sucessoPct: comparaveis ? (sucesso/comparaveis*100) : null, avaliadasCount: av.length, comparaveis };
      }

      function statusCampanhasResumo(campanhas){
        const grupos = { escalar:{count:0,soma:0,temSoma:false}, manter:{count:0,soma:0,temSoma:false}, ajustar:{count:0,soma:0,temSoma:false}, pausar:{count:0,soma:0,temSoma:false} };

          const g = grupos[c.classificacaoCodigo];
          if(!g) return;
          g.count++;
          if(c.margemDepoisDoAds !== null && c.margemDepoisDoAds !== undefined){ g.soma += Number(c.margemDepoisDoAds); g.temSoma = true; }
        });
        return grupos;

      function resumoIaHTML(data){
        const campanhas = data.campanhas || [];

        if(g.pausar.count) partes.push(g.pausar.count+' está'+(g.pausar.count===1?'':'ão')+' consumindo lucro');
        let texto = partes.length ? ('A IA encontrou: '+partes.join(', ')+'.') : 'Nenhuma campanha em situação crítica agora — todas dentro da margem mínima configurada.';

      }

      function exportarCsv(){
        const data = state.adsData;
        if(!data || !data.linhas || !data.linhas.length){ toast('Nada para exportar ainda.', 'error'); return; }
        const cab = ['Anuncio','SKU','Campanha','Investido','Vendas Ads (R$)','ROAS','Margem antes Ads (%)','Margem real (%)','Lucro (R$)','Classificacao IA'];

          return campos.map((v)=> '"'+String(v===null||v===undefined?'':v).replace(/"/g,'""')+'"').join(';');

        const blob = new Blob(['﻿'+csv], { type:'text/csv;charset=utf-8;' });

        setTimeout(()=> URL.revokeObjectURL(url), 2000);
      }

      // ---- Modal "Ver diagnóstico completo" (21/09/2026, pedido explícito
      // do usuário) — renderiza o relatório vindo do backend (texto com
      // parágrafos, uma tabela "| Métrica | Resultado | Leitura |" em
      // Markdown e uma lista de faixas do teste de monitoramento).
      // Conversor de Markdown propositalmente pequeno e local (só parágrafo/
      // tabela/lista — o único formato que lib/ia/adsDiagnostico.js produz),
      // nunca uma biblioteca nova (projeto sem acesso ao npm registry neste
      // ambiente de desenvolvimento, ver docs/05-problemas-conhecidos.md).
      function mdSimplesParaHtml(texto){
        const linhasTexto = String(texto||'').split('\n');
        let html = '';
        let i = 0;
        while(i < linhasTexto.length){
          const linha = linhasTexto[i];
          if(/^\s*\|.*\|\s*$/.test(linha)){
            const linhasTabela = [];
            while(i < linhasTexto.length && /^\s*\|.*\|\s*$/.test(linhasTexto[i])){ linhasTabela.push(linhasTexto[i]); i++; }
            const celulas = (l)=> l.trim().replace(/^\|/,'').replace(/\|$/,'').split('|').map(c=>c.trim());
            const cabecalho = celulas(linhasTabela[0]);
            const corpo = linhasTabela.slice(1).filter(l=> !/^[\s|:-]+$/.test(l)).map(celulas);
            html += '<div class="table-wrap" style="margin:10px 0;"><table class="data-table"><thead><tr>'
              + cabecalho.map(c=>'<th>'+escapeHtml(c)+'</th>').join('') + '</tr></thead><tbody>'
              + corpo.map(cols=>'<tr>'+cols.map(c=>'<td>'+escapeHtml(c)+'</td>').join('')+'</tr>').join('')
              + '</tbody></table></div>';
            continue;
          }
          if(/^\s*-\s+/.test(linha)){
            const itens = [];
            while(i < linhasTexto.length && /^\s*-\s+/.test(linhasTexto[i])){ itens.push(linhasTexto[i].replace(/^\s*-\s+/,'')); i++; }
            html += '<ul style="margin:8px 0;padding-left:20px;">' + itens.map(it=>'<li style="margin-bottom:4px;">'+escapeHtml(it)+'</li>').join('') + '</ul>';
            continue;
          }
          if(!linha.trim()){ i++; continue; }
          html += '<p style="margin:0 0 10px;line-height:1.55;">'+escapeHtml(linha)+'</p>';
          i++;
        }
        return html;
      }

      function approvalCardHTML(d){
        let acoesBtns = '<button class="promoc-btn approve" data-acao="aprovada" data-id="'+d.id+'" type="button">'+svg('check')+'<span>Aprovar</span></button>';
        if(podeAlterarNumerico(d)) acoesBtns += '<button class="promoc-btn" data-acao="alterar-toggle" data-id="'+d.id+'" type="button">'+svg('edit')+'<span>Alterar</span></button>';
        acoesBtns += '<button class="promoc-btn reject" data-acao="recusada" data-id="'+d.id+'" type="button">'+svg('x')+'<span>Recusar</span></button>';

        let html = '<article class="promoc-card">'
          + '<div class="promoc-card-top">'
            + '<div><div class="promoc-updated">Atualizado em '+fmtDataHora(d.atualizadoEm)+'</div>'
              + '<div class="promoc-title" title="'+escapeHtml(d.titulo||'—')+'">'+escapeHtml(d.titulo || '(sem título)')+'</div>'
              + '<div class="promoc-meta">SKU: '+escapeHtml(d.sku || 'sem SKU')+(d.promotionLabel ? ' &nbsp;|&nbsp; '+escapeHtml(d.promotionLabel) : '')+(d.loja ? ' &nbsp;|&nbsp; '+escapeHtml(d.loja) : '')+'</div>'
            + '</div>'
            + decisionBadgeHTML(d)
          + '</div>'
          + reasonGridHTML(d)
          + '<div class="promoc-compare-actions">'
            + compareHTML(d)
            + '<div class="promoc-actions">'+acoesBtns+'</div>'
          + '</div>';

        if(state.editandoId === d.id){
          html += '<div style="padding:0 14px 14px;">'
            + '<div class="inline-form" style="margin:0;align-items:flex-end;">'
              + '<div class="field narrow"><label>'+campoAlteracaoLabel(d)+'</label><input type="number" step="0.01" min="0" id="valorAlterado-'+d.id+'" value="'+escapeHtml(String(valorInicialAlteracao(d)))+'"></div>'
              + '<button class="primary-btn enabled" data-acao="confirmar-alterar" data-id="'+d.id+'" type="button">'+svg('check')+'<span>Confirmar alteração</span></button>'
              + '<button class="ghost-btn" data-acao="cancelar-alterar" data-id="'+d.id+'" type="button" style="cursor:pointer;opacity:1;"><span>Cancelar</span></button>'
            + '</div>'
          + '</div>';
        }
        return html + '</article>';
      }

      // ---- Painel lateral: linha do tempo real do 1º item da aba atual + resumo real do agente ----
      function timelineHTML(){
        const itemRef = state.itens[0];
        if(!itemRef) return '<div class="promoc-muted">Nenhum item pra mostrar a linha do tempo agora.</div><div class="promoc-timeline"></div>';
        const eventos = [];
        eventos.push({ titulo:'Identificado pela IA', data: itemRef.criadoEm, desc:'A IA coletou preço, desconto, margem e estoque reais do Mercado Livre pra este item.' });
        if(itemRef.atualizadoEm && itemRef.atualizadoEm !== itemRef.criadoEm) eventos.push({ titulo:'Última atualização', data: itemRef.atualizadoEm, desc:'Os números foram recalculados no ciclo automático mais recente.' });
        if(itemRef.decididoEm) eventos.push({ titulo:'Sua decisão registrada', data: itemRef.decididoEm, desc:(STATUS_LABEL[itemRef.statusDecisao]||itemRef.statusDecisao)+(itemRef.decididoPor? ' por '+itemRef.decididoPor : '') });
        if(itemRef.resultadoAvaliadoEm) eventos.push({ titulo:'Resultado reavaliado pela IA', data: itemRef.resultadoAvaliadoEm, desc:'Margem real depois da decisão: '+pctFmt(itemRef.resultadoSnapshot && itemRef.resultadoSnapshot.margemRealPct) });
        else if(itemRef.decididoEm) eventos.push({ titulo:'Aguardando reavaliação', data:null, desc:'A IA reavalia o resultado automaticamente alguns dias depois da decisão.' });

        return '<div class="promoc-muted" title="'+escapeHtml(itemRef.titulo||'')+'">'+escapeHtml(itemRef.titulo || '(sem título)')+'</div>'
          + '<div class="promoc-timeline">'
            + eventos.map((e)=> '<div class="promoc-event"><b>'+(e.data ? fmtDataHora(e.data)+' — ' : '')+escapeHtml(e.titulo)+'</b><p>'+escapeHtml(e.desc)+'</p></div>').join('')
          + '</div>';
      }

      function resumoLateralHTML(r){
        return '<div class="promoc-muted">Números reais de todas as decisões já registradas por este agente.</div>'
          + '<div class="promoc-summary-grid">'
            + '<div class="promoc-summary-box"><small>Aprovadas</small><b class="g">'+r.aprovadas+'</b></div>'
            + '<div class="promoc-summary-box"><small>Recusadas</small><b class="r">'+r.recusadas+'</b></div>'
            + '<div class="promoc-summary-box"><small>Decididas hoje</small><b>'+r.decididasHoje+'</b></div>'
            + '<div class="promoc-summary-box"><small>Já reavaliadas</small><b>'+r.avaliadasCount+'</b></div>'
          + '</div>'
          + '<div class="promoc-note">'+(r.avaliadasCount
              ? '💡 Margem média real nas '+r.avaliadasCount+' decisão(ões) já reavaliada(s) pela IA: '+pctFmt(r.margemMedia)+'.'
              : '💡 A IA reavalia o resultado real de cada decisão automaticamente alguns dias depois — ainda não há nenhuma reavaliação concluída.')+'</div>';
      }

      // ---- Histórico continua em tabela (mesmo formato de sempre) — só os cards de resumo acima mudaram ----
      function referenciaPromocaoHTML(d){
        return '<div class="cell-stack"><span class="prod-title" title="'+escapeHtml(d.titulo||'—')+'">'+escapeHtml(d.titulo || '(sem título)')+'</span>'
          + '<span class="cell-stack-sub mono">'+escapeHtml(d.sku || 'sem SKU')+'</span>'
          + '<span class="cell-stack-sub">'+escapeHtml(d.promotionLabel || d.promotionType || '')+'</span></div>';
      }
      function resultadoPromocaoHTML(d){
        const r = d.resultadoSnapshot || {};
        return '<div class="mono" style="'+(r.margemRealPct!==null && r.margemRealPct!==undefined ? (r.margemRealPct>=0?'color:var(--success);':'color:var(--danger);') : '')+'">Margem real: '+pctFmt(r.margemRealPct)+'</div>'
          + '<div class="cell-stack-sub">'+fmtDataHora(d.resultadoAvaliadoEm)+'</div>';
      }
      function tabelaHistoricoPromocoesHTML(itens){
        return '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>Anúncio / Promoção</th><th>Sugestão da IA (situação e motivo)</th><th>Minha decisão</th><th>Ação executada</th><th>Resultado posterior</th>'
          + '</tr></thead><tbody>'
          + itens.map((d)=> '<tr>'
            + '<td>'+referenciaPromocaoHTML(d)+'</td>'
            + '<td style="max-width:380px;">'+sugestaoResumoHTML(d)+'</td>'
            + '<td>'+decisaoResumoHTML(d)+'</td>'
            + '<td>'+executadoHTML(d, null)+'</td>'
            + '<td>'+(d.resultadoAvaliadoEm ? resultadoPromocaoHTML(d) : '<span style="color:var(--text-faint);font-style:italic;" title="A IA reavalia automaticamente alguns dias após a decisão.">Ainda não avaliado</span>')+'</td>'
          + '</tr>').join('')
          + '</tbody></table></div></div>';
      }

      function render(){
        const root = document.getElementById('agentePromocoesRoot');
        if(!root) return;
        const filtro = window.CerneFiltro.state;

        if(!filtro.loadingEmpresas && !filtro.empresas.length){
          root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
            + '<h2>Nenhuma empresa ativa</h2><p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> para usar este agente.</p></div>';
          return;
        }

        const r = resumoReal();
        const a = state.agente;
        let html = '<div class="table-card" style="padding:20px;margin-bottom:14px;display:flex;gap:16px;align-items:flex-start;flex-wrap:wrap;">'
          + '<div class="kpi-hero-icon" style="flex-shrink:0;">'+svg((a && a.icone) || 'sparkle')+'</div>'
          + '<div style="min-width:0;flex:1;">'
            + '<div class="panel-title" style="margin:0 0 4px;">'+escapeHtml((a && a.nome) || 'Promoções')+'</div>'
            + '<div class="field-hint" style="max-width:760px;">'+escapeHtml((a && a.descricao) || 'Analisa promoções do Mercado Livre e recomenda entrar, não entrar, sair ou revisar o preço promocional, sempre pela margem real.')+'</div>'
            + '<div class="field-hint" style="margin-top:8px;color:var(--text-faint);">Fase 1: a IA só analisa e recomenda — nenhuma ação é executada automaticamente no Mercado Livre. Toda decisão sua fica registrada no histórico.</div>'
          + '</div>'
        + '</div>';

        html += kpisHTML(r);

        html += '<div class="toolbar" style="margin-top:14px;">'
          + '<div class="seg" id="agentePromocoesTabs">'
            + '<button type="button" data-aba="pendentes" class="'+(state.aba==='pendentes'?'active':'')+'">Pendentes'+(r.pendentes ? ' ('+r.pendentes+')' : '')+'</button>'
            + '<button type="button" data-aba="historico" class="'+(state.aba==='historico'?'active':'')+'">Histórico</button>'
          + '</div>'
          + '<button class="ghost-btn" id="agentePromocoesGerarAgora" type="button" style="cursor:pointer;opacity:1;" '+(state.gerando?'disabled':'')+'>'+svg('refreshCw')+'<span>'+(state.gerando?'Gerando…':'Gerar sugestões agora')+'</span></button>'
        + '</div>';

        if(state.loading){
          html += '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando…</span></div></div>';
        } else if(state.error){
          html += '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
            + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p></div>';
        } else if(!state.itens.length){
          html += '<div class="empty-panel"><div class="empty-icon">'+svg('checkCircle')+'</div>'
            + '<h2>'+(state.aba==='pendentes' ? 'Nenhuma sugestão pendente' : 'Nenhuma decisão no histórico ainda')+'</h2>'
            + '<p class="lead">'+(state.aba==='pendentes'
                ? 'Quando a IA identificar uma promoção pra analisar, ela aparece aqui para você aprovar, alterar ou recusar.'
                : 'Decisões que você aprovar, alterar ou recusar aparecem aqui, junto do resultado real assim que a IA reavaliar.')+'</p></div>';
        } else if(state.aba === 'pendentes'){
          html += '<div class="promoc-workspace">'
            + '<div class="promoc-approvals">'+state.itens.map(approvalCardHTML).join('')+'</div>'
            + '<aside class="promoc-side">'
              + '<section class="promoc-side-card"><h3>Linha do tempo</h3>'+timelineHTML()+'</section>'
              + '<section class="promoc-side-card"><h3>Resumo do agente</h3>'+resumoLateralHTML(r)+'</section>'
            + '</aside>'
          + '</div>';
        } else {
          html += tabelaHistoricoPromocoesHTML(state.itens);
        }

        root.innerHTML = html;

        const tabsEl = document.getElementById('agentePromocoesTabs');
        if(tabsEl) tabsEl.querySelectorAll('button').forEach((b)=> b.addEventListener('click', ()=>{ state.aba = b.dataset.aba; state.editandoId = null; carregarDecisoes(); }));
        const btnGerar = document.getElementById('agentePromocoesGerarAgora');
        if(btnGerar) btnGerar.addEventListener('click', gerarAgora);

        root.querySelectorAll('[data-acao]').forEach((btn)=> btn.addEventListener('click', ()=>{
          const id = Number(btn.dataset.id);
          const acao = btn.dataset.acao;
          if(acao === 'alterar-toggle'){ state.editandoId = state.editandoId === id ? null : id; render(); return; }
          if(acao === 'cancelar-alterar'){ state.editandoId = null; render(); return; }
          if(acao === 'confirmar-alterar'){
            const input = document.getElementById('valorAlterado-'+id);
            const valor = Number(input.value);
            if(!Number.isFinite(valor) || valor < 0){ toast('Informe um valor válido.', 'error'); return; }
            const d = state.itens.find((x)=> x.id === id);
            if(!d) return;
            decidir(id, 'alterada', { [campoAlteracaoChave(d)]: valor });
            return;
          }
          if(acao === 'aprovada'){
            const d = state.itens.find((x)=> x.id === id);
            decidir(id, 'aprovada', d ? d.valorSugeridoIa : null);
            return;
          }
          if(acao === 'recusada'){ decidir(id, 'recusada', null); return; }
        }));
      }

      return { load };
    })();
  })();
