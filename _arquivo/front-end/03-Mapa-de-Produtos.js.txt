  /* ================= MAPA DE PRODUTOS — produto base + vínculo de SKU
     (19/09/2026, pedido explícito do usuário) =================
     A Estoque Full (valor em R$/caixas) e o Relatório de Produtos "Por
     Caixa" precisam saber, pra cada SKU de venda (kit, ex:
     '100CX-19X12X12'), qual é o produto FÍSICO real por trás dele
     (produto base, ex: 'CX-19X12X12') e quantas caixas físicas cada
     unidade desse kit representa (multiplicador). A API já existia
     inteira (routes/produtosBase.js) mas nunca teve tela — esta tela só
     consome essa API, nenhuma regra nova é criada aqui.
     Custo do produto base = custo por CAIXA física (nunca por kit) — é
     isso que a Estoque Full usa pra calcular o valor do estoque. Como o
     usuário normalmente só sabe o custo do KIT que comprou (ex: um kit de
     25 caixas), o formulário abaixo tem uma calculadora opcional: custo do
     kit ÷ caixas no kit = custo por caixa — preenche o campo de custo
     sozinha, mas o campo continua editável (o cálculo é só uma ajuda,
     nunca a única forma de preencher). */
  (function(){
    const state = {
      empresas:[], empresaId:null, loadingEmpresas:false,
      produtosBase:[], loading:false, error:null, status:'ativos', search:'',
      vinculos:[], loadingVinculos:false, errorVinculos:null,
      sugestoes:[], loadingSugestoes:false, errorSugestoes:null,
    };

    function toast(message, type){
      const wrap = document.getElementById('toastWrap');
      const el = document.createElement('div');
      el.className = 'toast ' + (type||'success');
      el.innerHTML = svg(type==='error' ? 'alertCircle':'checkCircle') + '<span>'+message+'</span>';
      wrap.appendChild(el);
      setTimeout(()=>{ el.style.opacity='0'; el.style.transition='opacity .25s'; setTimeout(()=>el.remove(),250); }, 3200);
    }
    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
    function money(v){ return v===null||v===undefined ? '—' : 'R$ ' + Number(v).toLocaleString('pt-BR',{minimumFractionDigits:2, maximumFractionDigits:2}); }
    // Entrada de valor decimal em pt-BR ("22,75" -> 22.75). Remove ponto de
    // milhar antes de trocar a vírgula decimal, pra aceitar "1.234,56" também.
    function parseDecimalBR(v){
      const s = String(v===undefined||v===null?'':v).trim();
      if(!s) return null;
      const normalizado = s.replace(/\./g,'').replace(',', '.');
      const n = Number(normalizado);
      return Number.isFinite(n) ? n : NaN;
    }
    function fmtDecimalInput(v){ return v===null||v===undefined ? '' : String(v).replace('.',','); }

    async function apiEmpresasAtivas(){
      const res = await fetch('/api/empresas?status=ativas');
      return ((await res.json()).empresas) || [];
    }
    async function api(path, opts){
      const res = await fetch('/api/produtos-base'+path, { headers:{'Content-Type':'application/json'}, ...opts });
      let body = null; try{ body = await res.json(); }catch(e){}
      if(!res.ok){ const err = new Error((body&&(body.error||(body.errors&&Object.values(body.errors)[0])))||'Falha na requisição.'); err.errors = body&&body.errors; throw err; }
      return body;
    }

    async function loadProdutosBase(){
      state.loading = true; state.error = null; render();
      try{
        const params = new URLSearchParams({ empresaId: state.empresaId });
        if(state.status !== 'todos') params.set('status', state.status);
        if(state.search.trim()) params.set('search', state.search.trim());
        const data = await api('?'+params.toString(), { method:'GET' });
        state.produtosBase = data.produtosBase;
      }catch(err){ state.error = err.message || 'Não foi possível carregar os produtos base.'; }
      finally{ state.loading = false; render(); }
    }
    async function loadVinculos(){
      state.loadingVinculos = true; state.errorVinculos = null; render();
      try{
        const data = await api('/vinculos?empresaId='+state.empresaId, { method:'GET' });
        state.vinculos = data.vinculos;
      }catch(err){ state.errorVinculos = err.message || 'Não foi possível carregar os vínculos.'; }
      finally{ state.loadingVinculos = false; render(); }
    }
    async function loadSugestoes(){
      state.loadingSugestoes = true; state.errorSugestoes = null; render();
      try{
        const data = await api('/vinculos/sugestoes?empresaId='+state.empresaId, { method:'GET' });
        state.sugestoes = data.sugestoes;
      }catch(err){ state.errorSugestoes = err.message || 'Não foi possível carregar as sugestões.'; }
      finally{ state.loadingSugestoes = false; render(); }
    }

    async function load(){
      state.loadingEmpresas = true; render();
      try{
        state.empresas = await apiEmpresasAtivas();
        if(!state.empresaId || !state.empresas.find(e=>e.id===state.empresaId)){
          state.empresaId = state.empresas.length ? state.empresas[0].id : null;
        }
      }catch(e){ state.empresas = []; }
      state.loadingEmpresas = false;
      if(state.empresaId){ await Promise.all([loadProdutosBase(), loadVinculos(), loadSugestoes()]); }
      else render();
    }

    function produtoBaseLabel(id){
      const p = state.produtosBase.find(x=>x.id===id);
      return p ? p.codigo : '—';
    }

    function tabelaProdutosBaseHTML(){
      if(state.loading){
        return '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando produtos base…</span></div></div>';
      }
      if(state.error){
        return '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
          + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.error)+'</p>'
          + '<button class="ghost-btn" id="mpRetry" style="cursor:pointer; opacity:1;">'+svg('activity')+'<span>Tentar novamente</span></button></div>';
      }
      if(!state.produtosBase.length){
        const filtrado = !!(state.search || state.status!=='ativos');
        return '<div class="empty-panel"><div class="empty-icon">'+svg('link')+'</div>'
          + '<h2>Nenhum produto base encontrado</h2>'
          + '<p class="lead">'+(filtrado ? 'Nenhum produto base corresponde ao filtro atual.' : 'Cadastre o primeiro produto base (o modelo físico real, ex: uma caixa 19x12x12) desta empresa.')+'</p>'
          + (filtrado ? '' : '<button class="primary-btn enabled" id="mpEmptyNew" type="button">'+svg('plus')+'<span>Novo produto base</span></button>')
        + '</div>';
      }
      return '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
        + '<th>Código</th><th>Nome</th><th>Custo por caixa</th><th>Status</th><th></th>'
        + '</tr></thead><tbody>'
        + state.produtosBase.map(p => '<tr class="'+(p.ativo?'':'inactive')+'">'
          + '<td class="mono">'+escapeHtml(p.codigo)+'</td>'
          + '<td>'+escapeHtml(p.nome||'—')+'</td>'
          + '<td class="tab-nums">'+(p.custo!==null ? money(p.custo) : '<span style="color:var(--text-faint); font-style:italic;">Pendente</span>')+'</td>'
          + '<td>'+(p.ativo ? '<span class="badge-status on"><span class="dot"></span>Ativo</span>' : '<span class="badge-status off"><span class="dot"></span>Inativo</span>')+'</td>'
          + '<td><div class="row-actions">'
            + '<button class="row-ic-btn" data-edit-base="'+p.id+'" title="Editar" type="button">'+svg('edit')+'</button>'
            + '<button class="row-ic-btn '+(p.ativo?'danger':'success')+'" data-toggle-base="'+p.id+'" data-next="'+(!p.ativo)+'" title="'+(p.ativo?'Desativar':'Ativar')+'" type="button">'+svg('power')+'</button>'
          + '</div></td>'
        + '</tr>').join('')
        + '</tbody></table></div></div>';
    }

    function tabelaVinculosHTML(){
      let html = '<div class="section-label" style="margin:18px 0 8px;"><span>Vínculos: SKU de venda → produto base</span></div>';
      if(state.loadingVinculos){
        return html + '<div class="table-card"><div class="table-skel"><span class="spin"></span><span>Carregando vínculos…</span></div></div>';
      }
      if(state.errorVinculos){
        return html + '<div class="empty-panel"><div class="empty-icon" style="color:var(--danger); background:var(--danger-soft);">'+svg('alertCircle')+'</div>'
          + '<h2 style="color:var(--danger);">Não foi possível carregar</h2><p class="lead">'+escapeHtml(state.errorVinculos)+'</p></div>';
      }
      if(!state.vinculos.length){
        html += '<div class="empty-panel"><div class="empty-icon">'+svg('tag')+'</div>'
          + '<h2>Nenhum vínculo cadastrado ainda</h2>'
          + '<p class="lead">Vincule um SKU de venda a um produto base (veja as sugestões abaixo, se houver) para que ele entre nas contas de Estoque Full e do Relatório "Por Caixa".</p></div>';
      } else {
        html += '<div class="table-card"><div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>SKU de venda</th><th>Produto base</th><th>Caixas por kit</th><th>Origem</th><th></th>'
          + '</tr></thead><tbody>'
          + state.vinculos.map(v => '<tr>'
            + '<td class="mono">'+escapeHtml(v.sku)+'</td>'
            + '<td class="mono">'+escapeHtml(v.produtoBaseCodigo||'—')+(v.produtoBaseNome?' <span style="color:var(--text-faint);">('+escapeHtml(v.produtoBaseNome)+')</span>':'')+'</td>'
            + '<td class="tab-nums">'+v.multiplicador+'</td>'
            + '<td>'+(v.origem==='automatico' ? '<span class="badge-status warn"><span class="dot"></span>Automático</span>' : '<span class="badge-status on"><span class="dot"></span>Manual</span>')+'</td>'
            + '<td><div class="row-actions">'
              + '<button class="row-ic-btn" data-edit-vinculo="'+v.id+'" title="Editar" type="button">'+svg('edit')+'</button>'
              + '<button class="row-ic-btn danger" data-remove-vinculo="'+v.id+'" title="Remover" type="button">'+svg('trash')+'</button>'
            + '</div></td>'
          + '</tr>').join('')
          + '</tbody></table></div></div>';
      }
      return html;
    }

    function painelSugestoesHTML(){
      if(state.loadingSugestoes || state.errorSugestoes || !state.sugestoes.length) return '';
      return '<div class="section-label" style="margin:18px 0 8px;"><span>SKUs vendidos ainda sem vínculo</span></div>'
        + '<div class="empty-panel" style="padding:0; text-align:left;">'
        + '<div class="table-wrap"><table class="data-table"><thead><tr>'
          + '<th>SKU</th><th>Vezes vendido</th><th>Sugestão</th><th></th>'
          + '</tr></thead><tbody>'
          + state.sugestoes.slice(0,50).map(s => '<tr>'
            + '<td class="mono">'+escapeHtml(s.sku)+'</td>'
            + '<td class="tab-nums">'+s.ocorrencias+'</td>'
            + '<td>'+(s.sugestao
                ? ('<span class="mono">'+escapeHtml(s.sugestao.codigoBase)+'</span> × '+s.sugestao.multiplicador+' caixas'+(s.sugestao.produtoBaseExistenteId?'':' <span style="color:var(--text-faint);">(produto base novo)</span>'))
                : '<span style="color:var(--text-faint); font-style:italic;">Sem sugestão — o SKU não segue o padrão "dígitos + código"</span>')
              +'</td>'
            + '<td><button class="row-ic-btn" data-vincular-sugestao="'+escapeHtml(s.sku)+'" title="Vincular" type="button">'+svg('link')+'</button></td>'
          + '</tr>').join('')
          + '</tbody></table></div></div>'
        + (state.sugestoes.length>50 ? '<div class="field-hint" style="margin-top:8px;">Mostrando 50 de '+state.sugestoes.length+' SKUs sem vínculo.</div>' : '');
    }

    function render(){
      const root = document.getElementById('mapaProdutosRoot');
      if(!root) return;

      if(!state.loadingEmpresas && !state.empresas.length){
        root.innerHTML = '<div class="empty-panel"><div class="empty-icon">'+svg('building')+'</div>'
          + '<h2>Nenhuma empresa ativa</h2>'
          + '<p class="lead">Cadastre e ative uma empresa em <strong>Cadastros → Empresas</strong> antes de cadastrar o mapa de produtos.</p></div>';
        return;
      }

      let html = '<div class="toolbar">'
        + '<div class="empresa-picker" style="margin-bottom:0;"><select id="mpEmpresaSelect">'
          + state.empresas.map(e=>'<option value="'+e.id+'" '+(e.id===state.empresaId?'selected':'')+'>'+escapeHtml(e.nomeFantasia||e.razaoSocial)+'</option>').join('')
        + '</select></div>'
        + '<div class="seg" id="mpSeg">'
          + [['ativos','Ativos'],['inativos','Inativos'],['todos','Todos']].map(function(par){ return '<button data-status="'+par[0]+'" class="'+(state.status===par[0]?'active':'')+'">'+par[1]+'</button>'; }).join('')
        + '</div>'
        + '<div class="search-box">'+svg('search')+'<input type="text" id="mpSearch" placeholder="Buscar por código ou nome" value="'+state.search.replace(/"/g,'&quot;')+'"></div>'
      + '</div>';

      html += tabelaProdutosBaseHTML();
      html += tabelaVinculosHTML();
      html += '<div style="margin:14px 0;"><button class="ghost-btn" id="mpNovoVinculo" style="cursor:pointer; opacity:1;" type="button">'+svg('plus')+'<span>Novo vínculo</span></button></div>';
      html += painelSugestoesHTML();

      root.innerHTML = html;

      const sel = document.getElementById('mpEmpresaSelect');
      if(sel) sel.addEventListener('change', ()=>{ state.empresaId = Number(sel.value); load(); });
      const seg = document.getElementById('mpSeg');
      if(seg) seg.querySelectorAll('button').forEach(b=> b.addEventListener('click', ()=>{ state.status = b.dataset.status; loadProdutosBase(); }));
      const search = document.getElementById('mpSearch');
      if(search) search.addEventListener('input', ()=>{
        state.search = search.value;
        clearTimeout(search._t);
        search._t = setTimeout(loadProdutosBase, 300);
      });
      const retry = document.getElementById('mpRetry');
      if(retry) retry.addEventListener('click', loadProdutosBase);
      const emptyNew = document.getElementById('mpEmptyNew');
      if(emptyNew) emptyNew.addEventListener('click', ()=> openModalBase());
      root.querySelectorAll('[data-edit-base]').forEach(b=> b.addEventListener('click', ()=>{
        const p = state.produtosBase.find(x=> String(x.id)===b.dataset.editBase);
        openModalBase(p);
      }));
      root.querySelectorAll('[data-toggle-base]').forEach(b=> b.addEventListener('click', async ()=>{
        const id = b.dataset.toggleBase, next = b.dataset.next === 'true';
        b.disabled = true;
        try{
          await api('/'+id+'/status', { method:'PATCH', body: JSON.stringify({ ativo: next }) });
          toast(next ? 'Produto base ativado.' : 'Produto base desativado.');
          loadProdutosBase();
        }catch(err){ toast(err.message, 'error'); b.disabled = false; }
      }));
      root.querySelectorAll('[data-edit-vinculo]').forEach(b=> b.addEventListener('click', ()=>{
        const v = state.vinculos.find(x=> String(x.id)===b.dataset.editVinculo);
        openModalVinculo(v);
      }));
      root.querySelectorAll('[data-remove-vinculo]').forEach(b=> b.addEventListener('click', async ()=>{
        b.disabled = true;
        try{
          await api('/vinculos/'+b.dataset.removeVinculo, { method:'DELETE' });
          toast('Vínculo removido.');
          loadVinculos(); loadSugestoes();
        }catch(err){ toast(err.message, 'error'); b.disabled = false; }
      }));
      const novoVinculo = document.getElementById('mpNovoVinculo');
      if(novoVinculo) novoVinculo.addEventListener('click', ()=> openModalVinculo());
      root.querySelectorAll('[data-vincular-sugestao]').forEach(b=> b.addEventListener('click', ()=>{
        const s = state.sugestoes.find(x=> x.sku===b.dataset.vincularSugestao);
        openModalVinculo(null, s);
      }));
    }

    // ---- Modal: Produto base (com calculadora "custo do kit ÷ caixas no kit") ----
    function openModalBase(produtoBase){
      const isEdit = !!produtoBase;
      const modalRoot = document.getElementById('modalRoot');
      modalRoot.innerHTML = '<div class="modal-overlay" id="modalOverlay">'
        + '<div class="modal-card">'
          + '<div class="modal-head"><div class="modal-title">'+(isEdit?'Editar produto base':'Novo produto base')+'</div>'
            + '<button class="row-ic-btn" id="modalClose" type="button">'+svg('x')+'</button></div>'
          + '<div class="modal-sub">'+(isEdit?'Atualize o código, nome e custo por caixa.':'Cadastre o modelo físico real (ex: uma caixa 19x12x12) que os SKUs de venda vão apontar.')+'</div>'
          + '<form id="mpFormBase">'
            + '<div class="field"><label>Código</label><input type="text" id="fCodigo" placeholder="Ex: CX-19X12X12" value="'+(isEdit?escapeHtml(produtoBase.codigo):'')+'" maxlength="100">'
              + '<div class="field-error" id="eCodigo" style="display:none;"></div></div>'
            + '<div class="field"><label>Nome <span class="opt">(opcional)</span></label><input type="text" id="fNome" placeholder="Ex: Caixa de papelão 19x12x12" value="'+(isEdit?escapeHtml(produtoBase.nome||''):'')+'" maxlength="200"></div>'
            + '<div class="field-hint">Não sabe o custo por caixa de cabeça? Preencha o custo do kit que você comprou e quantas caixas ele tem — calculamos pra você (mas o campo abaixo continua editável direto, se preferir).</div>'
            + '<div style="display:flex; gap:10px;">'
              + '<div class="field" style="flex:1;"><label>Custo do kit (R$) <span class="opt">(opcional)</span></label><input type="text" inputmode="decimal" id="fCustoKit" placeholder="Ex: 22,75"></div>'
              + '<div class="field" style="flex:1;"><label>Caixas no kit <span class="opt">(opcional)</span></label><input type="number" min="1" step="1" id="fCaixasKit" placeholder="Ex: 25"></div>'
            + '</div>'
            + '<div class="field"><label>Custo por caixa (R$) <span class="opt">(opcional)</span></label><input type="text" inputmode="decimal" id="fCusto" placeholder="Ex: 0,91" value="'+(isEdit?fmtDecimalInput(produtoBase.custo):'')+'">'
              + '<div class="field-hint" id="hintCustoCalc"></div>'
              + '<div class="field-error" id="eCusto" style="display:none;"></div></div>'
            + (isEdit ? '<div class="toggle-row"><div><div class="toggle-row-label">Produto base ativo</div><div class="toggle-row-sub">Produtos base inativos ficam ocultos nas seleções do sistema.</div></div>'
              + '<button type="button" class="toggle '+(produtoBase.ativo?'on':'')+'" id="fAtivoBase" data-on="'+(produtoBase.ativo?'true':'false')+'"></button></div>' : '')
            + '<div class="modal-actions">'
              + '<button type="button" class="btn-secondary" id="modalCancel">Cancelar</button>'
              + '<button type="submit" class="btn-primary" id="modalSave">'+svg('check')+'<span>'+(isEdit?'Salvar alterações':'Cadastrar produto base')+'</span></button>'
            + '</div>'
          + '</form>'
        + '</div>'
      + '</div>';

      const overlay = document.getElementById('modalOverlay');
      const close = ()=> modalRoot.innerHTML = '';
      overlay.addEventListener('click', (e)=>{ if(e.target === overlay) close(); });
      document.getElementById('modalClose')?.addEventListener('click', close);
      document.getElementById('modalCancel')?.addEventListener('click', close);
      document.getElementById('fCodigo').focus();

      const toggleBtn = document.getElementById('fAtivoBase');
      if(toggleBtn) toggleBtn.addEventListener('click', ()=>{
        const on = toggleBtn.dataset.on !== 'true';
        toggleBtn.dataset.on = String(on);
        toggleBtn.classList.toggle('on', on);
      });

      // Calculadora: custo do kit ÷ caixas no kit = custo por caixa. Só
      // preenche automaticamente enquanto o usuário não tiver editado o
      // campo de custo na mão nesta mesma abertura do modal — nunca
      // sobrescreve um valor que o usuário acabou de digitar direto.
      const custoKitInput = document.getElementById('fCustoKit');
      const caixasKitInput = document.getElementById('fCaixasKit');
      const custoInput = document.getElementById('fCusto');
      const hintCalc = document.getElementById('hintCustoCalc');
      let custoEditadoManualmente = false;
      custoInput.addEventListener('input', ()=>{ custoEditadoManualmente = true; hintCalc.textContent = ''; });
      function recalcular(){
        const custoKit = parseDecimalBR(custoKitInput.value);
        const caixasKit = Number(caixasKitInput.value);
        if(custoKit === null || !caixasKit){ hintCalc.textContent=''; return; }
        if(!Number.isFinite(custoKit) || custoKit < 0 || !Number.isInteger(caixasKit) || caixasKit <= 0){
          hintCalc.textContent = 'Informe um custo e uma quantidade de caixas válidos para calcular.';
          return;
        }
        const porCaixa = custoKit / caixasKit;
        hintCalc.textContent = money(custoKit)+' ÷ '+caixasKit+' caixas = '+money(porCaixa)+' por caixa.';
        if(!custoEditadoManualmente){
          custoInput.value = fmtDecimalInput(Math.round(porCaixa*100)/100);
        }
      }
      custoKitInput.addEventListener('input', recalcular);
      caixasKitInput.addEventListener('input', recalcular);

      document.getElementById('mpFormBase')?.addEventListener('submit', async (e)=>{
        e.preventDefault();
        ['eCodigo','eCusto'].forEach(id=>{ const el=document.getElementById(id); el.style.display='none'; el.textContent=''; });
        ['fCodigo','fCusto'].forEach(id=> document.getElementById(id).classList.remove('invalid'));

        const custoDigitado = custoInput.value.trim();
        const custo = custoDigitado ? parseDecimalBR(custoDigitado) : null;
        if(custo !== null && !Number.isFinite(custo)){
          document.getElementById('eCusto').textContent = 'Custo por caixa inválido.';
          document.getElementById('eCusto').style.display = 'block';
          document.getElementById('fCusto').classList.add('invalid');
          return;
        }

        const payload = {
          empresaId: state.empresaId,
          codigo: document.getElementById('fCodigo').value,
          nome: document.getElementById('fNome').value,
          custo,
        };
        if(isEdit && toggleBtn) payload.ativo = toggleBtn.dataset.on === 'true';

        const saveBtn = document.getElementById('modalSave');
        saveBtn.disabled = true;
        try{
          if(isEdit) await api('/'+produtoBase.id, { method:'PUT', body: JSON.stringify(payload) });
          else await api('', { method:'POST', body: JSON.stringify(payload) });
          close();
          toast(isEdit ? 'Produto base atualizado.' : 'Produto base cadastrado.');
          loadProdutosBase();
        }catch(err){
          saveBtn.disabled = false;
          if(err.errors){
            const fieldMap = { codigo:'Codigo', custo:'Custo' };
            Object.entries(err.errors).forEach(([k,msg])=>{
              const suf = fieldMap[k]; if(!suf) return;
              const el=document.getElementById('e'+suf); if(el){ el.textContent=msg; el.style.display='block'; }
              const input=document.getElementById('f'+suf); if(input) input.classList.add('invalid');
            });
          } else {
            toast(err.message, 'error');
          }
        }
      });
    }

    // ---- Modal: Vínculo (SKU -> produto base + multiplicador) ----
    // `sugestao` (opcional): { sku, sugestao:{multiplicador, codigoBase, produtoBaseExistenteId} } vinda do
    // painel "SKUs vendidos ainda sem vínculo" — só pré-preenche o formulário, o usuário sempre confirma.
    function openModalVinculo(vinculo, sugestao){
      const isEdit = !!vinculo;
      const baseSugeridaId = sugestao && sugestao.sugestao ? sugestao.sugestao.produtoBaseExistenteId : null;
      const multiplicadorSugerido = sugestao && sugestao.sugestao ? sugestao.sugestao.multiplicador : null;
      const modalRoot = document.getElementById('modalRoot');

      // Só produtos base ativos entram na lista — exceto o produto base
      // ATUAL do vínculo sendo editado, mesmo se foi desativado depois:
      // senão o select "perderia" a seleção e um salvamento sem querer
      // trocaria o vínculo pra outro produto base.
      const basesParaListar = state.produtosBase.filter(p=> p.ativo || (isEdit && vinculo.produtoBaseId===p.id));
      const opcoesBase = basesParaListar.map(p=>
        '<option value="'+p.id+'" '+((isEdit && vinculo.produtoBaseId===p.id) || (!isEdit && baseSugeridaId===p.id) ?'selected':'')+'>'+escapeHtml(p.codigo)+(p.nome?' — '+escapeHtml(p.nome):'')+(!p.ativo?' (inativo)':'')+'</option>'
      ).join('');

      modalRoot.innerHTML = '<div class="modal-overlay" id="modalOverlay">'
        + '<div class="modal-card">'
          + '<div class="modal-head"><div class="modal-title">'+(isEdit?'Editar vínculo':'Novo vínculo')+'</div>'
            + '<button class="row-ic-btn" id="modalClose" type="button">'+svg('x')+'</button></div>'
          + '<div class="modal-sub">Diz ao ERP quantas caixas de qual produto base cada unidade deste SKU de venda representa.</div>'
          + '<form id="mpFormVinculo">'
            + '<div class="field"><label>SKU de venda</label><input type="text" id="fSku" placeholder="Ex: 100CX-19X12X12" value="'+(isEdit?escapeHtml(vinculo.sku):(sugestao?escapeHtml(sugestao.sku):''))+'" maxlength="100" '+(isEdit?'disabled':'')+'>'
              + (isEdit ? '<div class="field-hint">O SKU de um vínculo já salvo não pode ser trocado — remova e crie um novo, se precisar.</div>' : '')
              + '<div class="field-error" id="eSku" style="display:none;"></div></div>'
            + '<div class="field"><label>Produto base</label>'
              + (opcoesBase ? '<select id="fProdutoBase"><option value="">Selecione…</option>'+opcoesBase+'</select>'
                            : '<div class="field-hint" style="margin-top:0;">Nenhum produto base ativo cadastrado ainda — cadastre um primeiro (botão "Novo produto base" no topo da tela).</div>')
              + (!isEdit && sugestao && sugestao.sugestao && !baseSugeridaId ? '<div class="field-hint">Sugestão: <strong>'+escapeHtml(sugestao.sugestao.codigoBase)+'</strong> (ainda não cadastrado — cadastre-o primeiro, depois volte aqui).</div>' : '')
              + '<div class="field-error" id="eProdutoBase" style="display:none;"></div></div>'
            + '<div class="field"><label>Caixas por kit (multiplicador)</label><input type="number" min="1" step="1" id="fMultiplicador" placeholder="Ex: 100" value="'+(isEdit?vinculo.multiplicador:(multiplicadorSugerido||''))+'">'
              + '<div class="field-hint">Quantas caixas físicas cada unidade deste SKU representa (ex: um anúncio "100CX-19X12X12" disponível = 100 caixas).</div>'
              + '<div class="field-error" id="eMultiplicador" style="display:none;"></div></div>'
            + '<div class="modal-actions">'
              + '<button type="button" class="btn-secondary" id="modalCancel">Cancelar</button>'
              + '<button type="submit" class="btn-primary" id="modalSave" '+(opcoesBase?'':'disabled')+'>'+svg('check')+'<span>'+(isEdit?'Salvar alterações':'Criar vínculo')+'</span></button>'
            + '</div>'
          + '</form>'
        + '</div>'
      + '</div>';

      const overlay = document.getElementById('modalOverlay');
      const close = ()=> modalRoot.innerHTML = '';
      overlay.addEventListener('click', (e)=>{ if(e.target === overlay) close(); });
      document.getElementById('modalClose')?.addEventListener('click', close);
      document.getElementById('modalCancel')?.addEventListener('click', close);
      document.getElementById(isEdit?'fMultiplicador':'fSku').focus();

      document.getElementById('mpFormVinculo')?.addEventListener('submit', async (e)=>{
        e.preventDefault();
        ['eSku','eProdutoBase','eMultiplicador'].forEach(id=>{ const el=document.getElementById(id); el.style.display='none'; el.textContent=''; });
        ['fSku','fMultiplicador'].forEach(id=>{ const el=document.getElementById(id); if(el) el.classList.remove('invalid'); });

        const produtoBaseSelect = document.getElementById('fProdutoBase');
        const payload = {
          empresaId: state.empresaId,
          multiplicador: Number(document.getElementById('fMultiplicador').value),
        };
        if(!isEdit) payload.sku = document.getElementById('fSku').value;
        if(produtoBaseSelect) payload.produtoBaseId = produtoBaseSelect.value ? Number(produtoBaseSelect.value) : null;

        const saveBtn = document.getElementById('modalSave');
        saveBtn.disabled = true;
        try{
          if(isEdit) await api('/vinculos/'+vinculo.id, { method:'PUT', body: JSON.stringify(payload) });
          else await api('/vinculos', { method:'POST', body: JSON.stringify(payload) });
          close();
          toast(isEdit ? 'Vínculo atualizado.' : 'Vínculo criado.');
          loadVinculos(); loadSugestoes();
        }catch(err){
          saveBtn.disabled = false;
          if(err.errors){
            const fieldMap = { sku:'Sku', produtoBaseId:'ProdutoBase', produtoBase:'ProdutoBase', multiplicador:'Multiplicador' };
            Object.entries(err.errors).forEach(([k,msg])=>{
              const suf = fieldMap[k]; if(!suf) return;
              const el=document.getElementById('e'+suf); if(el){ el.textContent=msg; el.style.display='block'; }
              const input=document.getElementById('f'+suf); if(input) input.classList.add('invalid');
            });
          } else {
            toast(err.message, 'error');
          }
        }
      });
    }

    document.getElementById('btnNewProdutoBase')?.addEventListener('click', ()=> openModalBase());

    window.MapaProdutos = { load };
  })();
